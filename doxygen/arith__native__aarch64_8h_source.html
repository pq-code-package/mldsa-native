49/*
50 * Sampling 256 coefficients mod 15 using rejection sampling from 4 bits.
51 * Expected number of required bytes: (256 * (16/15))/2 = 136.5 bytes.
52 * We sample 1 block (=136 bytes) of SHAKE256_RATE output initially.
53 * Sampling 2 blocks initially results in slightly worse performance.
54 */
55#define MLD_AARCH64_REJ_UNIFORM_ETA2_BUFLEN (1 * 136)
56/*
57 * Sampling 256 coefficients mod 9 using rejection sampling from 4 bits.
58 * Expected number of required bytes: (256 * (16/9))/2 = 227.5 bytes.
59 * We sample 2 blocks (=272 bytes) of SHAKE256_RATE output initially.
60 */
61#define MLD_AARCH64_REJ_UNIFORM_ETA4_BUFLEN (2 * 136)

63#define mld_ntt_aarch64_asm MLD_NAMESPACE(ntt_aarch64_asm)
64void mld_ntt_aarch64_asm(int32_t r[MLDSA_N], const int32_t zetas_l123456[144],
65 const int32_t zetas_l78[384])
66/* This must be kept in sync with the HOL-Light specification
67 * in proofs/hol_light/aarch64/proofs/mldsa_ntt_aarch64_asm.ml */
68__contract__(
69 requires(memory_no_alias(r, sizeof(int32_t) * MLDSA_N))
70 /* check-magic: 8380417 == MLDSA_Q */
71 requires(array_abs_bound(r, 0, MLDSA_N, 8380417))
72 requires(zetas_l123456 == mld_aarch64_ntt_zetas_layer123456)
73 requires(zetas_l78 == mld_aarch64_ntt_zetas_layer78)
74 assigns(memory_slice(r, sizeof(int32_t) * MLDSA_N))
75 /* check-magic: off */
76 ensures(array_abs_bound(r, 0, MLDSA_N, 75423753))
77 /* check-magic: on */
