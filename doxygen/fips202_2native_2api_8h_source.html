19/* Backends must return MLD_NATIVE_FUNC_SUCCESS upon success. */
20#define MLD_NATIVE_FUNC_SUCCESS (0)
21/* Backends may return MLD_NATIVE_FUNC_FALLBACK to signal to the frontend that
22 * the target/parameters are unsupported; typically, this would be because of
23 * dependencies on CPU features not detected on the host CPU. In this case,
24 * the frontend falls back to the default C implementation.
25 *
26 * IMPORTANT: Backend implementations must ensure that the decision of whether
27 * to fallback (return MLD_NATIVE_FUNC_FALLBACK) or not must never depend on
28 * the input data itself. Fallback decisions may only depend on system
29 * capabilities (e.g., CPU features) and, where present, length information.
30 * This requirement applies to all backend functions to maintain constant-time
31 * properties.
32 */
33#define MLD_NATIVE_FUNC_FALLBACK (-1)

35/*
36 * This is the C<->native interface allowing for the drop-in
37 * of custom Keccak-F1600 implementations.
38 *
39 * A _backend_ is a specific implementation of parts of this interface.
40 *
41 * You can replace 1-fold or 4-fold batched Keccak-F1600.
42 * To enable, set MLD_USE_NATIVE_FIPS202_X1 or MLD_USE_NATIVE_FIPS202_X4
43 * in your backend, and define the inline wrappers mld_keccak_f1600_x1_native()
44 * and/or mld_keccak_f1600_x4_native(), respectively, to forward to your
45 * implementation.
46 */
47
48#if defined(MLD_USE_NATIVE_FIPS202_X1)
49MLD_MUST_CHECK_RETURN_VALUE
50static MLD_INLINE int mld_keccak_f1600_x1_native(uint64_t *state)
51__contract__(
52 requires(memory_no_alias(state, sizeof(uint64_t) * 25 * 1))
53 assigns(memory_slice(state, sizeof(uint64_t) * 25 * 1))
54 ensures(return_value == MLD_NATIVE_FUNC_FALLBACK || return_value == MLD_NATIVE_FUNC_SUCCESS)

57#endif /* MLD_USE_NATIVE_FIPS202_X1 */

69/*
70 * Native x4 XOR bytes and extract bytes interface.
71 *
72 * These functions allow backends to provide optimized implementations for
73 * XORing input data into the state and extracting output data from the state.
74 * This is particularly useful for backends that use a different internal state
75 * representation (e.g., bit-interleaved), as conversion can happen during
76 * XOR/extract rather than before/after each permutation.
77 *
78 * NOTE: We assume that the custom representation of the zero state is the
79 * all-zero state.
80 *
81 * MLD_USE_NATIVE_FIPS202_X4_XOR_BYTES: Backend provides native XOR bytes
82 * MLD_USE_NATIVE_FIPS202_X4_EXTRACT_BYTES: Backend provides native extract
83 * bytes
84 */
85
86#if defined(MLD_USE_NATIVE_FIPS202_X4_XOR_BYTES)
87MLD_MUST_CHECK_RETURN_VALUE
88static MLD_INLINE int mld_keccakf1600_xor_bytes_x4_native(
89 uint64_t *state, const unsigned char *data0, const unsigned char *data1,
90 const unsigned char *data2, const unsigned char *data3, unsigned offset,
91 unsigned length)
92__contract__(
93 requires(0 <= offset && offset <= 25 * sizeof(uint64_t) &&
94 0 <= length && length <= 25 * sizeof(uint64_t) - offset)
95 requires(memory_no_alias(state, sizeof(uint64_t) * 25 * 4))
96 requires(memory_no_alias(data0, length))
97 requires((data0 == data1 &&
98 data0 == data2 &&
99 data0 == data3) ||
100 (memory_no_alias(data1, length) &&
101 memory_no_alias(data2, length) &&
102 memory_no_alias(data3, length)))
103 assigns(memory_slice(state, sizeof(uint64_t) * 25 * 4))
104 ensures(return_value == MLD_NATIVE_FUNC_FALLBACK || return_value == MLD_NATIVE_FUNC_SUCCESS)
105 ensures((return_value == MLD_NATIVE_FUNC_FALLBACK) ==> array_unchanged_u64(state, 25 * 4)));
106#endif /* MLD_USE_NATIVE_FIPS202_X4_XOR_BYTES */
