195/* Allocation macros for large local structures
196 *
197 * MLD_ALLOC(v, T, N) declares T *v and attempts to point it to an T[N]
198 * MLD_FREE(v, T, N) zeroizes and frees the allocation
199 *
200 * Default implementation uses stack allocation.
201 * Can be overridden by setting the config option MLD_CONFIG_CUSTOM_ALLOC_FREE
202 * and defining MLD_CUSTOM_ALLOC and MLD_CUSTOM_FREE.
203 */
204#if defined(MLD_CONFIG_CUSTOM_ALLOC_FREE) != \
205 (defined(MLD_CUSTOM_ALLOC) && defined(MLD_CUSTOM_FREE))
206#error Bad configuration: MLD_CONFIG_CUSTOM_ALLOC_FREE must be set together with MLD_CUSTOM_ALLOC and MLD_CUSTOM_FREE
207#endif
208
209/* Context-parameter machinery (MLD_CONTEXT_PARAMETERS_n and related config
210 * checks). Kept in a separate, level-generic header for readability; included
211 * here so it is available to the allocation macros below and to all consumers
212 * of common.h. */
213#include "context.h"
214
215#if !defined(MLD_CONFIG_CUSTOM_ALLOC_FREE)
216/* Default: stack allocation */
217
218/* This is a declaration macro, not an expression macro: T is a type and v is
219 * a declarator, neither of which can be wrapped in parentheses. The
220 * bugprone-macro-parentheses diagnostic is therefore a false positive here. */
221#define MLD_ALLOC(v, T, N, context) \
222 MLD_ALIGN T mld_alloc_##v[N]; \
223 T *v = mld_alloc_##v /* NOLINT(bugprone-macro-parentheses) */
224
225/* The MLD_FREE macro body references mld_zeroize(), which is declared in
226 * ct.h. We deliberately do NOT include ct.h here: doing so would create a
227 * circular dependency (ct.h includes common.h), and common.h itself never
228 * calls mld_zeroize() -- only the macro expansion does. Each translation
229 * unit that uses MLD_FREE therefore includes ct.h directly. */
230#define MLD_FREE(v, T, N, context) \
231 do \
232 { \
233 MLD_CONTEXT_UNUSED(context); \
234 mld_zeroize(mld_alloc_##v, sizeof(mld_alloc_##v)); \
235 (v) = NULL; \
236 } while (0)
237
238#else /* !MLD_CONFIG_CUSTOM_ALLOC_FREE */
239
240/* Custom allocation */
241
242/*
243 * The indirection here is necessary to use MLD_CONTEXT_PARAMETERS_3 here.
244 */
245#define MLD_APPLY(f, args) f args
246
247#define MLD_ALLOC(v, T, N, context) \
248 MLD_APPLY(MLD_CUSTOM_ALLOC, MLD_CONTEXT_PARAMETERS_3(v, T, N, context))
249
250#define MLD_FREE(v, T, N, context) \
251 do \
252 { \
253 if (v != NULL) \
254 { \
255 mld_zeroize(v, sizeof(T) * (N)); \
256 MLD_APPLY(MLD_CUSTOM_FREE, MLD_CONTEXT_PARAMETERS_3(v, T, N, context)); \
257 v = NULL; \
258 } \
259 } while (0)
260
261#endif /* MLD_CONFIG_CUSTOM_ALLOC_FREE */

263/****************************** Error codes ***********************************/
264
265/* Generic failure condition, reserved for failures not covered by a more
266 * specific error code. */
267#define MLD_ERR_FAIL (-1)
268/* An allocation failed. This can only happen if MLD_CONFIG_CUSTOM_ALLOC_FREE
269 * is defined and the provided MLD_CUSTOM_ALLOC can fail. */
270#define MLD_ERR_OUT_OF_MEMORY (-2)
271/* An RNG failure occurred. Might be due to insufficient entropy or
272 * system misconfiguration. */
273#define MLD_ERR_RNG_FAIL (-3)
274/* The signing rejection-sampling loop exceeded
275 * MLD_CONFIG_MAX_SIGNING_ATTEMPTS iterations without producing a valid
276 * signature. With a FIPS 204 Appendix C compliant bound (>= 821) this
277 * has probability < 2^-256. */
278#define MLD_ERR_SIGN_ATTEMPTS_EXHAUSTED (-4)
279/* Signing was paused before completing, at the request of a caller-provided
280 * MLD_CONFIG_SIGN_HOOK_ATTEMPT hook (see mldsa_native_config.h). The caller
281 * resumes by re-invoking signing with the same inputs; the attempt hook,
282 * together with MLD_CONFIG_SIGN_HOOK_RESUME, decides where to continue. */
283#define MLD_ERR_SIGNING_PAUSED (-5)
284/* Signature verification failed: the signature is not valid for the given
285 * message and public key. Returned by the verification API. */
286#define MLD_ERR_INVALID_SIGNATURE (-6)
287/* Secret key validation failed: the secret key is malformed or internally
288 * inconsistent. Returned by pk_from_sk. */
289#define MLD_ERR_INVALID_KEY (-7)
290/* The Pairwise Consistency Test failed. Only possible when
291 * MLD_CONFIG_KEYGEN_PCT is enabled; signals that the freshly generated key
292 * pair failed its sign/verify self-test. */
293#define MLD_ERR_PCT_FAIL (-8)
294/* An argument was invalid, e.g. an unsupported pre-hash algorithm or a context
295 * string longer than 255 bytes. */
296#define MLD_ERR_INVALID_ARG (-9)
