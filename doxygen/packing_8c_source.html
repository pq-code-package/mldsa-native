75MLD_INTERNAL_API
76int mld_pack_sig_h(uint8_t sig[MLDSA_CRYPTO_BYTES], const mld_polyveck *w0,
77 const mld_polyveck *w1)
78{
79 unsigned int j, k, n;
80
81 /* The hint section of sig[] is MLDSA_POLYVECH_PACKEDBYTES long, where
82 * MLDSA_POLYVECH_PACKEDBYTES = MLDSA_OMEGA + MLDSA_K.
83 *
84 * The first OMEGA bytes record the index numbers of the coefficients
85 * that are not equal to 0.
86 *
87 * The final K bytes record a running tally of the number of hints
88 * coming from each of the K polynomials. */
89 uint8_t *sig_h = sig + MLDSA_SIG_H_OFFSET;
90
91 mld_memset(sig_h, 0, MLDSA_POLYVECH_PACKEDBYTES);
92 n = 0;
93
94 /* For each coefficient of each polynomial, compute its hint bit and, if
95 * non-zero, record the index in the hint section of sig. If recording the
96 * hint would overflow the OMEGA-sized index array, abort early and return
97 * MLD_ERR_FAIL. The caller is expected to reject the signature in that case.
98 *
99 * Constant time: At this point w0/w1 are public (see comment in sign.c
100 * before the call), so a data-dependent early return is fine. */
101 for (k = 0; k < MLDSA_K; k++)
102 __loop__(
103 assigns(k, j, n, memory_slice(sig_h, MLDSA_POLYVECH_PACKEDBYTES))
104 invariant(k <= MLDSA_K && n <= MLDSA_OMEGA)
105 decreases(MLDSA_K - k)
106 )
107 {
108 for (j = 0; j < MLDSA_N; j++)
109 __loop__(
110 assigns(j, n, memory_slice(sig_h, MLDSA_POLYVECH_PACKEDBYTES))
111 invariant(j <= MLDSA_N && n <= MLDSA_OMEGA)
112 decreases(MLDSA_N - j)
113 )
114 {
115 const unsigned int hint_bit =
116 mld_make_hint(w0->vec[k].coeffs[j], w1->vec[k].coeffs[j]);
117 if (hint_bit)
118 {
119 if (n == MLDSA_OMEGA)
120 {
121 return MLD_ERR_FAIL;
122 }
123 /* Safety: branch above ensures n < MLDSA_OMEGA so n is a valid index
124 * into the OMEGA-sized index array; j < MLDSA_N <= 256 fits in
125 * uint8_t. */
126 sig_h[n] = (uint8_t)j;
127 n++;
128 }
129 }
130 /* Record the running tally into the correct slot for this polynomial.
131 * Safety: k < MLDSA_K, so MLDSA_OMEGA + k is a valid index into the
132 * K-byte tally tail; n <= MLDSA_OMEGA fits in uint8_t. */
133 sig_h[MLDSA_OMEGA + k] = (uint8_t)n;
134 }
135 return 0;
136}

147MLD_INTERNAL_API
148int mld_sig_unpack_hints(mld_poly *h, const uint8_t sig[MLDSA_CRYPTO_BYTES],
149 unsigned int i)
150{
151 const uint8_t *packed_hints = sig + MLDSA_SIG_H_OFFSET;
152 const unsigned int old_hint_count =
153 (i == 0) ? 0 : packed_hints[MLDSA_OMEGA + i - 1];
154 const unsigned int new_hint_count = packed_hints[MLDSA_OMEGA + i];
155 unsigned int j;
156
157 if (new_hint_count < old_hint_count || new_hint_count > MLDSA_OMEGA)
158 {
159 return MLD_ERR_FAIL;
160 }
161
162 mld_memset(h, 0, sizeof(mld_poly));
163
164 for (j = old_hint_count; j < new_hint_count; ++j)
165 __loop__(
166 invariant(j >= old_hint_count && j <= new_hint_count &&
167 new_hint_count <= MLDSA_OMEGA)
168 invariant(array_bound(h->coeffs, 0, MLDSA_N, 0, 2))
169 invariant(forall(p, 0, MLDSA_N,
170 (h->coeffs[p] == 1) ==
171 exists(hj, old_hint_count, j, packed_hints[hj] == p)))
172 decreases(new_hint_count - j)
173 )
174 {
175 if (j > old_hint_count && packed_hints[j] <= packed_hints[j - 1])
176 {
177 return MLD_ERR_FAIL;
178 }
179 /* Safety: packed_hints[j] is uint8_t (<= 255) and MLDSA_N == 256. */
180 h->coeffs[packed_hints[j]] = 1;
181 }
182
183 /* On the last row, also verify that the trailing index slots are zero. */
184 if (i == MLDSA_K - 1)
185 {
186 for (j = new_hint_count; j < MLDSA_OMEGA; ++j)
187 __loop__(
188 invariant(j <= MLDSA_OMEGA)
189 decreases(MLDSA_OMEGA - j)
190 )
191 {
192 if (packed_hints[j] != 0)
193 {
194 return MLD_ERR_FAIL;
195 }
196 }
197 }
198
199 /* On success, h->coeffs[p] is 1 exactly for the hint indices decoded for this
200 * row, i.e. packed_hints[old_hint_count, new_hint_count). Asserted here
201 * rather than posted as a contract to not unnecessarily increase proof
202 * complexity at callers that do not need the functional description. */
203 cassert(forall(
204 p, 0, MLDSA_N,
205 (h->coeffs[p] == 1) ==
206 exists(hj, old_hint_count, new_hint_count, packed_hints[hj] == p)));
207
208 return 0;
209}
210#endif /* !MLD_CONFIG_NO_VERIFY_API */
