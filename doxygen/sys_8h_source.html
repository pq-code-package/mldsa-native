220/* New X86_64 CPUs support control-flow protection using the CET instructions.
221 * When enabled (through -fcf-protection=), all compilation units (including
222 * empty ones) need to support CET for this to work.
223 * For assembly, this means that source files need to signal support for
224 * CET by setting the appropriate note.gnu.property section.
225 * This can be achieved by including the <cet.h> header in all assembly file.
226 * This file also provides the _CET_ENDBR macro which needs to be placed at
227 * every potential target of an indirect branch.
228 * If CET is enabled _CET_ENDBR maps to the endbr64 instruction, otherwise
229 * it is empty.
230 * In case the compiler does not support CET (e.g., <gcc8, <clang11),
231 * the __CET__ macro is not set and we default to nothing.
232 * Note that we only issue _CET_ENDBR instructions through the MLD_ASM_FN_SYMBOL
233 * macro as the global symbols are the only possible targets of indirect
234 * branches in our code.
235 */
236#if defined(MLD_SYS_X86_64)
237#if defined(__CET__)
238#include <cet.h>
239#define MLD_CET_ENDBR _CET_ENDBR
240#else
241#define MLD_CET_ENDBR
242#endif
243#endif /* MLD_SYS_X86_64 */

268/* The x86_64 assembly backend uses the SysV calling convention. On Windows,
269 * where the Microsoft x64 calling convention is the default, it can still be
270 * used with compilers that allow choosing the calling convention per
271 * function: GCC and Clang support __attribute__((sysv_abi)), which makes
272 * calls to the annotated function follow the SysV calling convention.
273 *
274 * MLD_SYSV_ABI_SUPPORTED signals that the toolchain can call SysV assembly
275 * routines; the x86_64 assembly backend is only enabled if it is defined.
276 * MLD_SYSV_ABI is the attribute carried by declarations of x86_64 assembly
277 * routines. Both macros can be set externally for toolchains offering an
278 * equivalent mechanism that is not recognized here. */
279#if defined(MLD_SYS_X86_64) && !defined(MLD_SYSV_ABI_SUPPORTED)
280#if !defined(MLD_SYS_WINDOWS) || defined(__GNUC__) || defined(__clang__)
281#define MLD_SYSV_ABI_SUPPORTED
282#endif
283#endif
