41MLD_STATIC_TESTABLE
42void mld_poly_decompose_c(mld_poly *a1, mld_poly *a0)
43__contract__(
44 requires(memory_no_alias(a1, sizeof(mld_poly)))
45 requires(memory_no_alias(a0, sizeof(mld_poly)))
46 requires(array_bound(a0->coeffs, 0, MLDSA_N, 0, MLDSA_Q))
47 assigns(memory_slice(a1, sizeof(mld_poly)))
48 assigns(memory_slice(a0, sizeof(mld_poly)))
49 ensures(array_bound(a1->coeffs, 0, MLDSA_N, 0, (MLDSA_Q-1)/(2*MLDSA_GAMMA2)))
50 ensures(array_abs_bound(a0->coeffs, 0, MLDSA_N, MLDSA_GAMMA2+1))
51)
52{
53 unsigned int i;
54 mld_assert_bound(a0->coeffs, MLDSA_N, 0, MLDSA_Q);
55 for (i = 0; i < MLDSA_N; ++i)
56 __loop__(
57 assigns(i, memory_slice(a0, sizeof(mld_poly)), memory_slice(a1, sizeof(mld_poly)))
58 invariant(i <= MLDSA_N)
59 invariant(array_bound(a0->coeffs, i, MLDSA_N, 0, MLDSA_Q))
60 invariant(array_bound(a1->coeffs, 0, i, 0, (MLDSA_Q-1)/(2*MLDSA_GAMMA2)))
61 invariant(array_abs_bound(a0->coeffs, 0, i, MLDSA_GAMMA2+1))
62 decreases(MLDSA_N - i)
63 )
64 {
65 mld_decompose(&a0->coeffs[i], &a1->coeffs[i], a0->coeffs[i]);
66 }
67
68 mld_assert_abs_bound(a0->coeffs, MLDSA_N, MLDSA_GAMMA2 + 1);
69 mld_assert_bound(a1->coeffs, MLDSA_N, 0, (MLDSA_Q - 1) / (2 * MLDSA_GAMMA2));
70}

180/* Reference: `mld_rej_eta()` in the reference implementation @[REF].
181 * - Our signature differs from the reference implementation
182 * in that it adds the offset and always expects the base of the
183 * target buffer. This avoids shifting the buffer base in the
184 * caller, which appears tricky to reason about. */
185#if MLDSA_ETA == 2
186/*
187 * Sampling 256 coefficients mod 15 using rejection sampling from 4 bits.
188 * Expected number of required bytes: (256 * (16/15))/2 = 136.5 bytes.
189 * We sample 1 block (=136 bytes) of SHAKE256_RATE output initially.
190 * Sampling 2 blocks initially results in slightly worse performance.
191 */
192#define MLD_POLY_UNIFORM_ETA_NBLOCKS 1
193#elif MLDSA_ETA == 4
194/*
195 * Sampling 256 coefficients mod 9 using rejection sampling from 4 bits.
196 * Expected number of required bytes: (256 * (16/9))/2 = 227.5 bytes.
197 * We sample 2 blocks (=272 bytes) of SHAKE256_RATE output initially.
198 */
199#define MLD_POLY_UNIFORM_ETA_NBLOCKS 2
200#else /* MLDSA_ETA == 4 */
201#error "Invalid value of MLDSA_ETA"
202#endif /* MLDSA_ETA != 2 && MLDSA_ETA != 4 */
203
204MLD_STATIC_TESTABLE unsigned int mld_rej_eta_c(int32_t *a, unsigned int target,
205 unsigned int offset,
206 const uint8_t *buf,
207 unsigned int buflen)
208__contract__(
209 requires(offset <= target && target <= MLDSA_N)
210 requires(buflen <= (MLD_POLY_UNIFORM_ETA_NBLOCKS * MLD_STREAM256_BLOCKBYTES))
211 requires(memory_no_alias(a, sizeof(int32_t) * target))
212 requires(memory_no_alias(buf, buflen))
213 requires(array_abs_bound(a, 0, offset, MLDSA_ETA + 1))
214 assigns(memory_slice(a, sizeof(int32_t) * target))
215 ensures(offset <= return_value && return_value <= target)
216 ensures(array_abs_bound(a, 0, return_value, MLDSA_ETA + 1))
217)
218{
219 unsigned int ctr, pos;
220 int t_valid;
221 uint32_t t0, t1;
222 mld_assert_abs_bound(a, offset, MLDSA_ETA + 1);
223 ctr = offset;
224 pos = 0;
225 while (ctr < target && pos < buflen)
226 __loop__(
227 invariant(offset <= ctr && ctr <= target && pos <= buflen)
228 invariant(array_abs_bound(a, 0, ctr, MLDSA_ETA + 1))
229 decreases(buflen - pos)
230 )
231 {
232 t0 = buf[pos] & 0x0F;
233 t1 = buf[pos++] >> 4;
234
235 /* Constant time: The inputs and outputs to the rejection sampling are
236 * secret. However, it is fine to leak which coefficients have been
237 * rejected. For constant-time testing, we declassify the result of
238 * the comparison.
239 */
240#if MLDSA_ETA == 2
241 t_valid = t0 < 15;
242 MLD_CT_TESTING_DECLASSIFY(&t_valid, sizeof(int));
243 if (t_valid) /* t0 < 15 */
244 {
245 t0 = t0 - (205 * t0 >> 10) * 5;
246 a[ctr++] = 2 - (int32_t)t0;
247 }
248 t_valid = t1 < 15;
249 MLD_CT_TESTING_DECLASSIFY(&t_valid, sizeof(int));
250 if (t_valid && ctr < target) /* t1 < 15 */
251 {
252 t1 = t1 - (205 * t1 >> 10) * 5;
253 a[ctr++] = 2 - (int32_t)t1;
254 }
255#elif MLDSA_ETA == 4
256 t_valid = t0 < 9;
257 MLD_CT_TESTING_DECLASSIFY(&t_valid, sizeof(int));
258 if (t_valid) /* t0 < 9 */
259 {
260 a[ctr++] = 4 - (int32_t)t0;
261 }
262 t_valid = t1 < 9; /* t1 < 9 */
263 MLD_CT_TESTING_DECLASSIFY(&t_valid, sizeof(int));
264 if (t_valid && ctr < target)
265 {
266 a[ctr++] = 4 - (int32_t)t1;
267 }
268#else /* MLDSA_ETA == 4 */
269#error "Invalid value of MLDSA_ETA"
270#endif /* MLDSA_ETA != 2 && MLDSA_ETA != 4 */
271 }
272
273 mld_assert_abs_bound(a, ctr, MLDSA_ETA + 1);
274
275 return ctr;
276}

278static unsigned int mld_rej_eta(int32_t *a, unsigned int target,
279 unsigned int offset, const uint8_t *buf,
280 unsigned int buflen)
281__contract__(
282 requires(offset <= target && target <= MLDSA_N)
283 requires(buflen <= (MLD_POLY_UNIFORM_ETA_NBLOCKS * MLD_STREAM256_BLOCKBYTES))
284 requires(memory_no_alias(a, sizeof(int32_t) * target))
285 requires(memory_no_alias(buf, buflen))
286 requires(array_abs_bound(a, 0, offset, MLDSA_ETA + 1))
287 assigns(memory_slice(a, sizeof(int32_t) * target))
288 ensures(offset <= return_value && return_value <= target)
289 ensures(array_abs_bound(a, 0, return_value, MLDSA_ETA + 1))
290)
291{
292#if MLDSA_ETA == 2 && defined(MLD_USE_NATIVE_REJ_UNIFORM_ETA2)
293 int ret;
294 mld_assert_abs_bound(a, offset, MLDSA_ETA + 1);
295 if (offset == 0)
296 {
297 ret = mld_rej_uniform_eta2_native(a, target, buf, buflen);
298 if (ret != MLD_NATIVE_FUNC_FALLBACK)
299 {
300 unsigned res = (unsigned)ret;
301 mld_assert_abs_bound(a, res, MLDSA_ETA + 1);
302 return res;
303 }
304 }
305#elif MLDSA_ETA == 4 && defined(MLD_USE_NATIVE_REJ_UNIFORM_ETA4)
306 int ret;
307 mld_assert_abs_bound(a, offset, MLDSA_ETA + 1);
308 if (offset == 0)
309 {
310 ret = mld_rej_uniform_eta4_native(a, target, buf, buflen);
311 if (ret != MLD_NATIVE_FUNC_FALLBACK)
312 {
313 unsigned res = (unsigned)ret;
314 mld_assert_abs_bound(a, res, MLDSA_ETA + 1);
315 return res;
316 }
317 }
318#endif /* !(MLDSA_ETA == 2 && MLD_USE_NATIVE_REJ_UNIFORM_ETA2) && MLDSA_ETA == \
319 4 && MLD_USE_NATIVE_REJ_UNIFORM_ETA4 */
320
321 return mld_rej_eta_c(a, target, offset, buf, buflen);
322}

325MLD_INTERNAL_API
326void mld_poly_uniform_eta_4x(mld_poly *r0, mld_poly *r1, mld_poly *r2,
327 mld_poly *r3, const uint8_t seed[MLDSA_CRHBYTES],
328 uint8_t nonce0, uint8_t nonce1, uint8_t nonce2,
329 uint8_t nonce3)
330{
331 /* Temporary buffers for XOF output before rejection sampling */
332 MLD_ALIGN uint8_t buf[4][MLD_ALIGN_UP(MLD_POLY_UNIFORM_ETA_NBLOCKS *
333 MLD_STREAM256_BLOCKBYTES)];
334
335 MLD_ALIGN uint8_t extseed[4][MLD_ALIGN_UP(MLDSA_CRHBYTES + 2)];
336
337 /* Tracks the number of coefficients we have already sampled */
338 unsigned ctr[4];
339 mld_xof256_x4_ctx state;
340 unsigned buflen;
341
342 mld_memcpy(extseed[0], seed, MLDSA_CRHBYTES);
343 mld_memcpy(extseed[1], seed, MLDSA_CRHBYTES);
344 mld_memcpy(extseed[2], seed, MLDSA_CRHBYTES);
345 mld_memcpy(extseed[3], seed, MLDSA_CRHBYTES);
346 extseed[0][MLDSA_CRHBYTES] = nonce0;
347 extseed[1][MLDSA_CRHBYTES] = nonce1;
348 extseed[2][MLDSA_CRHBYTES] = nonce2;
349 extseed[3][MLDSA_CRHBYTES] = nonce3;
350 extseed[0][MLDSA_CRHBYTES + 1] = 0;
351 extseed[1][MLDSA_CRHBYTES + 1] = 0;
352 extseed[2][MLDSA_CRHBYTES + 1] = 0;
353 extseed[3][MLDSA_CRHBYTES + 1] = 0;
354
355 mld_xof256_x4_init(&state);
356 mld_xof256_x4_absorb(&state, extseed, MLDSA_CRHBYTES + 2);
357
358 /*
359 * Initially, squeeze heuristic number of MLD_POLY_UNIFORM_ETA_NBLOCKS.
360 * This should generate the coefficients with high probability.
361 */
362 mld_xof256_x4_squeezeblocks(buf, MLD_POLY_UNIFORM_ETA_NBLOCKS, &state);
363 buflen = MLD_POLY_UNIFORM_ETA_NBLOCKS * MLD_STREAM256_BLOCKBYTES;
364
365 ctr[0] = mld_rej_eta(r0->coeffs, MLDSA_N, 0, buf[0], buflen);
366 ctr[1] = mld_rej_eta(r1->coeffs, MLDSA_N, 0, buf[1], buflen);
367 ctr[2] = mld_rej_eta(r2->coeffs, MLDSA_N, 0, buf[2], buflen);
368 ctr[3] = mld_rej_eta(r3->coeffs, MLDSA_N, 0, buf[3], buflen);
369
370 /*
371 * So long as not all entries have been generated, squeeze
372 * one more block at a time until we're done.
373 */
374 buflen = MLD_STREAM256_BLOCKBYTES;
375 while (ctr[0] < MLDSA_N || ctr[1] < MLDSA_N || ctr[2] < MLDSA_N ||
376 ctr[3] < MLDSA_N)
377 __loop__(
378 assigns(ctr, state, memory_slice(r0, sizeof(mld_poly)),
379 memory_slice(r1, sizeof(mld_poly)), memory_slice(r2, sizeof(mld_poly)),
380 memory_slice(r3, sizeof(mld_poly)), object_whole(buf[0]),
381 object_whole(buf[1]), object_whole(buf[2]),
382 object_whole(buf[3]))
383 invariant(ctr[0] <= MLDSA_N && ctr[1] <= MLDSA_N)
384 invariant(ctr[2] <= MLDSA_N && ctr[3] <= MLDSA_N)
385 invariant(array_abs_bound(r0->coeffs, 0, ctr[0], MLDSA_ETA + 1))
386 invariant(array_abs_bound(r1->coeffs, 0, ctr[1], MLDSA_ETA + 1))
387 invariant(array_abs_bound(r2->coeffs, 0, ctr[2], MLDSA_ETA + 1))
388 invariant(array_abs_bound(r3->coeffs, 0, ctr[3], MLDSA_ETA + 1)))
389 {
390 mld_xof256_x4_squeezeblocks(buf, 1, &state);
391 ctr[0] = mld_rej_eta(r0->coeffs, MLDSA_N, ctr[0], buf[0], buflen);
392 ctr[1] = mld_rej_eta(r1->coeffs, MLDSA_N, ctr[1], buf[1], buflen);
393 ctr[2] = mld_rej_eta(r2->coeffs, MLDSA_N, ctr[2], buf[2], buflen);
394 ctr[3] = mld_rej_eta(r3->coeffs, MLDSA_N, ctr[3], buf[3], buflen);
395 }
396
397 mld_xof256_x4_release(&state);
398
399 mld_assert_abs_bound(r0->coeffs, MLDSA_N, MLDSA_ETA + 1);
400 mld_assert_abs_bound(r1->coeffs, MLDSA_N, MLDSA_ETA + 1);
401 mld_assert_abs_bound(r2->coeffs, MLDSA_N, MLDSA_ETA + 1);
402 mld_assert_abs_bound(r3->coeffs, MLDSA_N, MLDSA_ETA + 1);
403
404 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
405 mld_zeroize(buf, sizeof(buf));
406 mld_zeroize(extseed, sizeof(extseed));
407}

410MLD_INTERNAL_API
411void mld_poly_uniform_eta(mld_poly *r, const uint8_t seed[MLDSA_CRHBYTES],
412 uint8_t nonce)
413{
414 /* Temporary buffer for XOF output before rejection sampling */
415 MLD_ALIGN uint8_t
416 buf[MLD_POLY_UNIFORM_ETA_NBLOCKS * MLD_STREAM256_BLOCKBYTES];
417 MLD_ALIGN uint8_t extseed[MLDSA_CRHBYTES + 2];
418
419 /* Tracks the number of coefficients we have already sampled */
420 unsigned ctr;
421 mld_xof256_ctx state;
422 unsigned buflen;
423
424 mld_memcpy(extseed, seed, MLDSA_CRHBYTES);
425 extseed[MLDSA_CRHBYTES] = nonce;
426 extseed[MLDSA_CRHBYTES + 1] = 0;
427
428 mld_xof256_init(&state);
429 mld_xof256_absorb_once(&state, extseed, MLDSA_CRHBYTES + 2);
430
431 /*
432 * Initially, squeeze heuristic number of MLD_POLY_UNIFORM_ETA_NBLOCKS.
433 * This should generate the coefficients with high probability.
434 */
435 mld_xof256_squeezeblocks(buf, MLD_POLY_UNIFORM_ETA_NBLOCKS, &state);
436 buflen = MLD_POLY_UNIFORM_ETA_NBLOCKS * MLD_STREAM256_BLOCKBYTES;
437
438 ctr = mld_rej_eta(r->coeffs, MLDSA_N, 0, buf, buflen);
439
440 /*
441 * So long as not all entries have been generated, squeeze
442 * one more block at a time until we're done.
443 */
444 buflen = MLD_STREAM256_BLOCKBYTES;
445 while (ctr < MLDSA_N)
446 __loop__(
447 assigns(ctr, object_whole(&state),
448 object_whole(buf), memory_slice(r, sizeof(mld_poly)))
449 invariant(ctr <= MLDSA_N)
450 invariant(state.pos <= SHAKE256_RATE)
451 invariant(array_abs_bound(r->coeffs, 0, ctr, MLDSA_ETA + 1)))
452 {
453 mld_xof256_squeezeblocks(buf, 1, &state);
454 ctr = mld_rej_eta(r->coeffs, MLDSA_N, ctr, buf, buflen);
455 }
456
457 mld_xof256_release(&state);
458
459 mld_assert_abs_bound(r->coeffs, MLDSA_N, MLDSA_ETA + 1);
460
461 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
462 mld_zeroize(buf, sizeof(buf));
463 mld_zeroize(extseed, sizeof(extseed));
464}

476MLD_INTERNAL_API
477void mld_poly_uniform_gamma1(mld_poly *a, const uint8_t seed[MLDSA_CRHBYTES],
478 uint16_t nonce)
479{
480 MLD_ALIGN uint8_t
481 buf[MLD_POLY_UNIFORM_GAMMA1_NBLOCKS * MLD_STREAM256_BLOCKBYTES];
482 MLD_ALIGN uint8_t extseed[MLDSA_CRHBYTES + 2];
483 mld_xof256_ctx state;

485 mld_memcpy(extseed, seed, MLDSA_CRHBYTES);
486 extseed[MLDSA_CRHBYTES] = (uint8_t)(nonce & 0xFF);
487 extseed[MLDSA_CRHBYTES + 1] = (uint8_t)(nonce >> 8);

561#if !defined(MLD_CONFIG_NO_SIGN_API) || !defined(MLD_CONFIG_NO_VERIFY_API)
562MLD_INTERNAL_API
563void mld_poly_challenge(mld_poly *c, const uint8_t seed[MLDSA_CTILDEBYTES])
564{
565 unsigned int i, j, pos;
566 uint64_t signs;
567 uint64_t offset;
568 MLD_ALIGN uint8_t buf[SHAKE256_RATE];
569 mld_shake256ctx state;
570
571 mld_shake256_init(&state);
572 mld_shake256_absorb(&state, seed, MLDSA_CTILDEBYTES);
573 mld_shake256_finalize(&state);
574 mld_shake256_squeeze(buf, SHAKE256_RATE, &state);
575
576 /* Convert the first 8 bytes of buf[] into an unsigned 64-bit value. */
577 /* Each bit of that dictates the sign of the resulting challenge value */
578 signs = 0;
579 for (i = 0; i < 8; ++i)
580 __loop__(
581 assigns(i, signs)
582 invariant(i <= 8)
583 decreases(8 - i)
584 )
585 {
586 signs |= (uint64_t)buf[i] << 8 * i;
587 }
588 pos = 8;
589
590 mld_memset(c, 0, sizeof(mld_poly));
591
592 for (i = MLDSA_N - MLDSA_TAU; i < MLDSA_N; ++i)
593 __loop__(
594 assigns(i, j, object_whole(buf), state, pos, memory_slice(c, sizeof(mld_poly)), signs)
595 invariant(i >= MLDSA_N - MLDSA_TAU)
596 invariant(i <= MLDSA_N)
597 invariant(pos >= 1)
598 invariant(pos <= SHAKE256_RATE)
599 invariant(array_bound(c->coeffs, 0, MLDSA_N, -1, 2))
600 invariant(state.pos <= SHAKE256_RATE)
601 decreases(MLDSA_N - i)
602 )
603 {
604 /* This loop terminates only probabilistically, hence no decreases
605 * clause. */
606 do
607 __loop__(
608 assigns(j, object_whole(buf), state, pos)
609 invariant(state.pos <= SHAKE256_RATE)
610 )
611 {
612 if (pos >= SHAKE256_RATE)
613 {
614 mld_shake256_squeeze(buf, SHAKE256_RATE, &state);
615 pos = 0;
616 }
617 j = buf[pos++];
618 } while (j > i);
619
620 c->coeffs[i] = c->coeffs[j];
621
622 /* Reference: Compute coefficient value here in two steps to */
623 /* avoid mixing unsigned and signed arithmetic with implicit */
624 /* conversions, and so that CBMC can keep track of ranges */
625 /* to complete type-safety proof here. */
626
627 /* The least-significant bit of signs tells us if we want -1 or +1 */
628 offset = 2 * (signs & 1);
629
630 /* offset has value 0 or 2 here, so (1 - (int32_t) offset) has
631 * value -1 or +1 */
632 c->coeffs[j] = 1 - (int32_t)offset;
633
634 /* Move to the next bit of signs for next time */
635 signs >>= 1;
636 }
637
638 mld_assert_bound(c->coeffs, MLDSA_N, -1, 2);
639 mld_shake256_release(&state);
640
641 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
642 mld_zeroize(buf, sizeof(buf));
643 mld_zeroize(&signs, sizeof(signs));
644}
