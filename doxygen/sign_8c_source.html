186 const uint8_t seed[MLDSA_CRHBYTES])
187__contract__(
188 requires(memory_no_alias(s1, sizeof(mld_polyvecl)))
189 requires(memory_no_alias(s2, sizeof(mld_polyveck)))
190 requires(memory_no_alias(seed, MLDSA_CRHBYTES))
191 assigns(object_whole(s1), object_whole(s2))
192 ensures(forall(l0, 0, MLDSA_L, array_abs_bound(s1->vec[l0].coeffs, 0, MLDSA_N, MLDSA_ETA + 1)))
193 ensures(forall(k0, 0, MLDSA_K, array_abs_bound(s2->vec[k0].coeffs, 0, MLDSA_N, MLDSA_ETA + 1)))
194)
195{
196/* Sample short vectors s1 and s2 */
197#if defined(MLD_CONFIG_SERIAL_FIPS202_ONLY)
198 int i;
199 uint16_t nonce = 0;
200 /* Safety: The nonces are at most 14 (MLDSA_L + MLDSA_K - 1), and, hence, the
201 * casts are safe. */
202 for (i = 0; i < MLDSA_L; i++)
203 {
204 mld_poly_uniform_eta(&s1->vec[i], seed, (uint8_t)(nonce + i));
205 }
206 for (i = 0; i < MLDSA_K; i++)
207 {
208 mld_poly_uniform_eta(&s2->vec[i], seed, (uint8_t)(nonce + MLDSA_L + i));
209 }
210#else /* MLD_CONFIG_SERIAL_FIPS202_ONLY */
211#if MLD_CONFIG_PARAMETER_SET == 44
212 mld_poly_uniform_eta_4x(&s1->vec[0], &s1->vec[1], &s1->vec[2], &s1->vec[3],
213 seed, 0, 1, 2, 3);
214 mld_poly_uniform_eta_4x(&s2->vec[0], &s2->vec[1], &s2->vec[2], &s2->vec[3],
215 seed, 4, 5, 6, 7);
216#elif MLD_CONFIG_PARAMETER_SET == 65
217 mld_poly_uniform_eta_4x(&s1->vec[0], &s1->vec[1], &s1->vec[2], &s1->vec[3],
218 seed, 0, 1, 2, 3);
219 mld_poly_uniform_eta_4x(&s1->vec[4], &s2->vec[0], &s2->vec[1],
220 &s2->vec[2] /* irrelevant */, seed, 4, 5, 6,
221 0xFF /* irrelevant */);
222 mld_poly_uniform_eta_4x(&s2->vec[2], &s2->vec[3], &s2->vec[4], &s2->vec[5],
223 seed, 7, 8, 9, 10);
224#elif MLD_CONFIG_PARAMETER_SET == 87
225 mld_poly_uniform_eta_4x(&s1->vec[0], &s1->vec[1], &s1->vec[2], &s1->vec[3],
226 seed, 0, 1, 2, 3);
227 mld_poly_uniform_eta_4x(&s1->vec[4], &s1->vec[5], &s1->vec[6],
228 &s2->vec[0] /* irrelevant */, seed, 4, 5, 6,
229 0xFF /* irrelevant */);
230 mld_poly_uniform_eta_4x(&s2->vec[0], &s2->vec[1], &s2->vec[2], &s2->vec[3],

233 seed, 11, 12, 13, 14);
234#endif /* MLD_CONFIG_PARAMETER_SET == 87 */
235#endif /* !MLD_CONFIG_SERIAL_FIPS202_ONLY */
236}

263static int mld_compute_pack_t0_t1(
264 uint8_t pk_t1[MLDSA_K * MLDSA_POLYT1_PACKEDBYTES],
265 uint8_t t0_packed[MLDSA_K * MLDSA_POLYT0_PACKEDBYTES],
266 const mld_polyvecl *s1hat, const mld_polyveck *s2,
267 const uint8_t rho[MLDSA_SEEDBYTES],
268 MLD_CONFIG_CONTEXT_PARAMETER_TYPE context)
269__contract__(
270 requires(memory_no_alias(pk_t1, MLDSA_K * MLDSA_POLYT1_PACKEDBYTES))
271 requires(memory_no_alias(t0_packed, MLDSA_K * MLDSA_POLYT0_PACKEDBYTES))
272 requires(memory_no_alias(s1hat, sizeof(mld_polyvecl)))
273 requires(memory_no_alias(s2, sizeof(mld_polyveck)))
274 requires(memory_no_alias(rho, MLDSA_SEEDBYTES))
275 requires(forall(l1, 0, MLDSA_L,
276 array_abs_bound(s1hat->vec[l1].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
277 requires(forall(k2, 0, MLDSA_K,
278 array_bound(s2->vec[k2].coeffs, 0, MLDSA_N,
279 MLD_POLYETA_UNPACK_LOWER_BOUND, MLDSA_ETA + 1)))
280 assigns(memory_slice(pk_t1, MLDSA_K * MLDSA_POLYT1_PACKEDBYTES))
281 assigns(memory_slice(t0_packed, MLDSA_K * MLDSA_POLYT0_PACKEDBYTES))
282 ensures(return_value == 0 || return_value == MLD_ERR_OUT_OF_MEMORY))
283{
284 unsigned int k;
285 int ret;
286 MLD_ALLOC(mat, mld_polymat, 1, context);
287 MLD_ALLOC(t0k, mld_poly, 1, context);
288 MLD_ALLOC(t1k, mld_poly, 1, context);
289
290 if (mat == NULL || t0k == NULL || t1k == NULL)
291 {
292 ret = MLD_ERR_OUT_OF_MEMORY;
293 goto cleanup;
294 }
295
296 /* Expand matrix */
297 mld_polyvec_matrix_expand(mat, rho);
298
299 for (k = 0; k < MLDSA_K; k++)
300 __loop__(
301 assigns(k, memory_slice(pk_t1, MLDSA_K * MLDSA_POLYT1_PACKEDBYTES),
302 memory_slice(t0_packed, MLDSA_K * MLDSA_POLYT0_PACKEDBYTES),
303 memory_slice(t0k, sizeof(mld_poly)),
304 memory_slice(t1k, sizeof(mld_poly))
305 MLD_IF_REDUCE_RAM(, memory_slice(mat, sizeof(mld_polymat))))
306 invariant(k <= MLDSA_K)
307 decreases(MLDSA_K - k)
308 )
309 {
310 /* t0k = (A * s1hat)_k in NTT domain */
311 mld_polyvec_matrix_pointwise_montgomery_row(t0k, mat, s1hat, k);
312
313 /* t0k = invNTT(t0k) */
314 mld_poly_invntt_tomont(t0k);
315
316 /* t0k += s2[k] */
317 mld_poly_add(t0k, &s2->vec[k]);
318
319 /* Reference: The following reduction is not present in the reference
320 * implementation. Omitting this reduction requires the output
321 * of the invntt to be small enough such that the addition of
322 * s2 does not result in absolute values >= MLDSA_Q. While our
323 * C, x86_64, and AArch64 invntt implementations produce small
324 * enough values for this to work out, it complicates the
325 * bounds reasoning. We instead add an additional reduction,
326 * and can consequently, relax the bounds requirements for the
327 * invntt.
328 */
329 mld_poly_reduce(t0k);
330
331 /* Decompose into t1[k] and t0[k] (in place into t0k). */
332 mld_poly_caddq(t0k);
333 mld_poly_power2round(t1k, t0k, t0k);
334
335 /* Pack t1[k] into pk and t0[k] into the t0 output buffer. */
336 mld_polyt1_pack(pk_t1 + k * MLDSA_POLYT1_PACKEDBYTES, t1k);
337 mld_polyt0_pack(t0_packed + k * MLDSA_POLYT0_PACKEDBYTES, t0k);
338 }
339
340 ret = 0;
341cleanup:
342 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
343 MLD_FREE(t1k, mld_poly, 1, context);
344 MLD_FREE(t0k, mld_poly, 1, context);
345 MLD_FREE(mat, mld_polymat, 1, context);
346 return ret;
347}

350MLD_EXTERNAL_API
351int mld_sign_keypair_internal(uint8_t pk[MLDSA_CRYPTO_PUBLICKEYBYTES],
352 uint8_t sk[MLDSA_CRYPTO_SECRETKEYBYTES],
353 const uint8_t seed[MLDSA_SEEDBYTES],
354 MLD_CONFIG_CONTEXT_PARAMETER_TYPE context)
355{
356 int ret;
357 const uint8_t *rho, *rhoprime, *key;
358
359 MLD_ALLOC(seedbuf, uint8_t, 2 * MLDSA_SEEDBYTES + MLDSA_CRHBYTES, context);
360 MLD_ALLOC(inbuf, uint8_t, MLDSA_SEEDBYTES + 2, context);
361 MLD_ALLOC(tr, uint8_t, MLDSA_TRBYTES, context);
362 MLD_ALLOC(s1, mld_polyvecl, 1, context);
363 MLD_ALLOC(s2, mld_polyveck, 1, context);
364
365 if (seedbuf == NULL || inbuf == NULL || tr == NULL || s1 == NULL ||
366 s2 == NULL)
367 {
368 ret = MLD_ERR_OUT_OF_MEMORY;
369 goto cleanup;
370 }
371
372 /* Get randomness for rho, rhoprime and key */
373 mld_memcpy(inbuf, seed, MLDSA_SEEDBYTES);
374 inbuf[MLDSA_SEEDBYTES + 0] = MLDSA_K;
375 inbuf[MLDSA_SEEDBYTES + 1] = MLDSA_L;
376 mld_shake256(seedbuf, 2 * MLDSA_SEEDBYTES + MLDSA_CRHBYTES, inbuf,
377 MLDSA_SEEDBYTES + 2);
378 rho = seedbuf;
379 rhoprime = rho + MLDSA_SEEDBYTES;
380 key = rhoprime + MLDSA_CRHBYTES;
381
382 /* Constant time: rho is part of the public key and, hence, public. */
383 MLD_CT_TESTING_DECLASSIFY(rho, MLDSA_SEEDBYTES);
384
385 /* Sample s1 and s2 */
386 mld_sample_s1_s2(s1, s2, rhoprime);
387
388 /* Pack s1 into sk before NTT */
389 mld_pack_sk_s1(sk, s1);
390
391 /* NTT s1 in place to use as s1hat */
392 mld_polyvecl_ntt(s1);
393
394 /* Pack rho into pk */
395 mld_memcpy(pk + MLDSA_PK_RHO_OFFSET, rho, MLDSA_SEEDBYTES);
396
397 /* Compute t = A*s1hat + s2 row by row, decompose into t1/t0, and pack
398 * t1 into pk and t0 directly into the t0 region of sk. */
399 ret = mld_compute_pack_t0_t1(pk + MLDSA_PK_T1_OFFSET, sk + MLDSA_SK_T0_OFFSET,
400 s1, s2, rho, context);
401 if (ret != 0)
402 {
403 goto cleanup;
404 }
405
406 /* Compute tr = H(pk) */
407 mld_shake256(tr, MLDSA_TRBYTES, pk, MLDSA_CRYPTO_PUBLICKEYBYTES);
408
409 /* Pack remaining secret key components (s1 and t0 already packed) */
410 mld_pack_sk_rho_key_tr_s2(sk, rho, tr, key, s2);
411
412 /* Constant time: pk is the public key, inherently public data */
413 MLD_CT_TESTING_DECLASSIFY(pk, MLDSA_CRYPTO_PUBLICKEYBYTES);
414
415cleanup:
416 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
417 MLD_FREE(s2, mld_polyveck, 1, context);
418 MLD_FREE(s1, mld_polyvecl, 1, context);
419 MLD_FREE(tr, uint8_t, MLDSA_TRBYTES, context);
420 MLD_FREE(inbuf, uint8_t, MLDSA_SEEDBYTES + 2, context);
421 MLD_FREE(seedbuf, uint8_t, 2 * MLDSA_SEEDBYTES + MLDSA_CRHBYTES, context);
422
423 /* Pairwise Consistency Test (PCT) @[FIPS140_3_IG, p.87] */
424 /* Do this after freeing all temporaries. */
425 if (ret == 0)
426 {
427 ret = mld_check_pct(pk, sk, context);
428 }
429
430 if (ret != 0)
431 {
432 /* Clear caller outputs on failure. */
433 mld_zeroize(pk, MLDSA_CRYPTO_PUBLICKEYBYTES);
434 mld_zeroize(sk, MLDSA_CRYPTO_SECRETKEYBYTES);
435 }
436
437 return ret;
438}

490static void mld_H(uint8_t *out, size_t outlen, const uint8_t *in1,
491 size_t in1len, const uint8_t *in2, size_t in2len,
492 const uint8_t *in3, size_t in3len)

495 requires(in2len <= MLD_MAX_BUFFER_SIZE)
496 requires(in3len <= MLD_MAX_BUFFER_SIZE)
497 requires(outlen <= 8 * SHAKE256_RATE /* somewhat arbitrary bound */)
498 requires(memory_no_alias(in1, in1len))
499 requires(in2len == 0 || memory_no_alias(in2, in2len))

514 mld_shake256_absorb(&state, in3, in3len);
515 }

518 mld_shake256_release(&state);
519

521 mld_zeroize(&state, sizeof(state));

549MLD_MUST_CHECK_RETURN_VALUE
550static int mld_compute_pack_z(uint8_t sig[MLDSA_CRYPTO_BYTES],
551 const mld_poly *cp, const mld_sk_s1hat *s1hat,
552 const mld_yvec *y, mld_poly *z, mld_poly *tmp)
553__contract__(
554 requires(memory_no_alias(sig, MLDSA_CRYPTO_BYTES))
555 requires(memory_no_alias(cp, sizeof(mld_poly)))
556 requires(memory_no_alias(s1hat, sizeof(mld_sk_s1hat)))
557 requires(memory_no_alias(y, sizeof(mld_yvec)))
558 requires(memory_no_alias(z, sizeof(mld_poly)))
559 requires(memory_no_alias(tmp, sizeof(mld_poly)))
560 requires(array_abs_bound(cp->coeffs, 0, MLDSA_N, MLD_NTT_BOUND))
561 MLD_IF_NOT_REDUCE_RAM(
562 requires(forall(k0, 0, MLDSA_L,
563 array_bound(y->vec.vec[k0].coeffs, 0, MLDSA_N, -(MLDSA_GAMMA1 - 1), MLDSA_GAMMA1 + 1)))
564 requires(forall(k1, 0, MLDSA_L, array_abs_bound(s1hat->vec.vec[k1].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
565 )
566 MLD_IF_REDUCE_RAM(
567 requires(memory_no_alias(s1hat->packed, MLDSA_L * MLDSA_POLYETA_PACKEDBYTES))
568 requires(memory_no_alias(y->rhoprime, MLDSA_CRHBYTES))
569 requires(y->kappa <= MLD_MAX_KAPPA)
570 )
571 assigns(memory_slice(sig, MLDSA_CRYPTO_BYTES))
572 assigns(memory_slice(z, sizeof(mld_poly)))
573 assigns(memory_slice(tmp, sizeof(mld_poly)))
574 ensures(return_value == 0 || return_value == MLD_ERR_FAIL ||
575 return_value == MLD_ERR_OUT_OF_MEMORY)
576)
577{
578 unsigned int i;
579 uint32_t z_invalid;
580 for (i = 0; i < MLDSA_L; i++)
581 __loop__(
582 assigns(i, memory_slice(z, sizeof(mld_poly)),
583 memory_slice(tmp, sizeof(mld_poly)),
584 memory_slice(sig, MLDSA_CRYPTO_BYTES))
585 invariant(i <= MLDSA_L)
586 decreases(MLDSA_L - i)
587 )
588 {
589 mld_sk_s1hat_get_poly(z, s1hat, i);
590 mld_poly_pointwise_montgomery(z, cp);
591 mld_poly_invntt_tomont(z);
592 mld_yvec_get_poly(tmp, y, i);
593 mld_poly_add(z, tmp);
594 mld_poly_reduce(z);
595
596 z_invalid = mld_poly_chknorm(z, MLDSA_GAMMA1 - MLDSA_BETA);
597 /* Constant time: It is fine (and prohibitively expensive to avoid)
598 * to leak the result of the norm check and which polynomial in z caused a
599 * rejection. It would even be okay to leak which coefficient led to
600 * rejection as the candidate signature will be discarded anyway.
601 * See Section 5.5 of @[Round3_Spec]. */
602 MLD_CT_TESTING_DECLASSIFY(&z_invalid, sizeof(uint32_t));
603 if (z_invalid)
604 {
605 return MLD_ERR_FAIL; /* reject */
606 }
607 /* If z is valid, then its coefficients are bounded by
608 * MLDSA_GAMMA1 - MLDSA_BETA. This will be needed below
609 * to prove the pre-condition of pack_sig_z() */
610 mld_assert_abs_bound(z, MLDSA_N, (MLDSA_GAMMA1 - MLDSA_BETA));
611
612 /* After the norm check, the distribution of each coefficient of z is
613 * independent of the secret key and it can, hence, be considered
614 * public. It is, hence, okay to immediately pack it into the user-provided
615 * signature buffer. */
616 mld_pack_sig_z(sig, z, i);
617 }
618 return 0;
619}

692MLD_MUST_CHECK_RETURN_VALUE
693/* NOLINTNEXTLINE(readability-function-cognitive-complexity) */
694static int mld_attempt_signature_generation(
695 uint8_t sig[MLDSA_CRYPTO_BYTES], const uint8_t *mu,
696 const uint8_t rhoprime[MLDSA_CRHBYTES], uint16_t kappa, mld_polymat *mat,
697 const mld_sk_s1hat *s1hat, const mld_sk_s2hat *s2hat,
698 const mld_sk_t0hat *t0hat, MLD_CONFIG_CONTEXT_PARAMETER_TYPE context)
699__contract__(
700 requires(memory_no_alias(sig, MLDSA_CRYPTO_BYTES))
701 requires(memory_no_alias(mu, MLDSA_CRHBYTES))
702 requires(memory_no_alias(rhoprime, MLDSA_CRHBYTES))
703 requires(memory_no_alias(mat, sizeof(mld_polymat)))
704 requires(memory_no_alias(s1hat, sizeof(mld_sk_s1hat)))
705 requires(memory_no_alias(s2hat, sizeof(mld_sk_s2hat)))
706 requires(memory_no_alias(t0hat, sizeof(mld_sk_t0hat)))
707 requires(kappa <= MLD_MAX_KAPPA)
708 MLD_IF_NOT_REDUCE_RAM(
709 requires(forall(k1, 0, MLDSA_K, forall(l1, 0, MLDSA_L,
710 array_bound(mat->vec[k1].vec[l1].coeffs, 0, MLDSA_N, 0, MLDSA_Q))))
711 requires(forall(k2, 0, MLDSA_K, array_abs_bound(t0hat->vec.vec[k2].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
712 requires(forall(k3, 0, MLDSA_L, array_abs_bound(s1hat->vec.vec[k3].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
713 requires(forall(k4, 0, MLDSA_K, array_abs_bound(s2hat->vec.vec[k4].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
714 )
715 MLD_IF_REDUCE_RAM(
716 requires(memory_no_alias(s1hat->packed, MLDSA_L * MLDSA_POLYETA_PACKEDBYTES))
717 requires(memory_no_alias(s2hat->packed, MLDSA_K * MLDSA_POLYETA_PACKEDBYTES))
718 requires(memory_no_alias(t0hat->packed, MLDSA_K * MLDSA_POLYT0_PACKEDBYTES))
719 )
720 assigns(memory_slice(sig, MLDSA_CRYPTO_BYTES))
721 MLD_IF_REDUCE_RAM(
722 assigns(memory_slice(mat, sizeof(mld_polymat)))
723 )
724 ensures(return_value == 0 || return_value == MLD_ERR_FAIL ||
725 return_value == MLD_ERR_OUT_OF_MEMORY)
726)
727{
728 unsigned int k;
729 uint32_t w0_invalid, h_invalid;
730 int ret;
731
732 typedef union
733 {
734 mld_polyveck w1;
735 mld_polyvecl tmp;
736 } w1tmp_u;
737 mld_polyveck *w1;
738 mld_polyvecl *tmp;
739
740 MLD_ALLOC(challenge_bytes, uint8_t, MLDSA_CTILDEBYTES, context);
741 MLD_ALLOC(y, mld_yvec, 1, context);
742 MLD_ALLOC(z, mld_poly, 1, context);
743 MLD_ALLOC(w1tmp, w1tmp_u, 1, context);
744 MLD_ALLOC(w0, mld_polyveck, 1, context);
745 MLD_ALLOC(cp, mld_poly, 1, context);
746 MLD_ALLOC(t, mld_poly, 1, context);
747
748 if (challenge_bytes == NULL || y == NULL || z == NULL || w1tmp == NULL ||
749 w0 == NULL || cp == NULL || t == NULL)
750 {
751 ret = MLD_ERR_OUT_OF_MEMORY;
752 goto cleanup;
753 }
754 w1 = &w1tmp->w1;
755 tmp = &w1tmp->tmp;
756
757 /* @[FIPS204, Algorithm 7, line 11] y <- ExpandMask(rhoprime, kappa). */
758 mld_yvec_init(y, rhoprime, kappa);
759
760 /* @[FIPS204, Algorithm 7, line 12] w <- invNTT(A_hat o NTT(y)). This call
761 * performs the whole line: it NTTs y, accumulates the pointwise product with
762 * A_hat, and applies the inverse NTT. In REDUCE_RAM mode the y sampling is
763 * fused into the same pass. */
764 mld_polyvec_matrix_pointwise_montgomery_yvec(w0, mat, y, tmp);
765
766 /* @[FIPS204, Algorithm 7, line 13] w1 <- HighBits(w), here together with the
767 * low part: Decompose yields w = 2*GAMMA2*w1 + w0, keeping both w1 and w0
768 * (w0 is reused below in the line-21/26 alternative, see further down). */
769 mld_polyveck_caddq(w0);
770 mld_polyveck_decompose(w1, w0);
771
772 /* @[FIPS204, Algorithm 7, line 15] ctilde <- H(mu || w1Encode(w1), lambda/4).
773 * w1Encode(w1) is packed into the w1 region of sig (mld_polyveck_pack_w1),
774 * then absorbed by H together with mu. */
775 mld_polyveck_pack_w1(sig, w1);
776
777 mld_H(challenge_bytes, MLDSA_CTILDEBYTES, mu, MLDSA_CRHBYTES, sig,
778 MLDSA_K * MLDSA_POLYW1_PACKEDBYTES, NULL, 0);
779 /* Constant time: Leaking challenge_bytes does not reveal any information
780 * about the secret key as H() is modelled as random oracle.
781 * This also applies to challenges for rejected signatures.
782 * See Section 5.5 of @[Round3_Spec]. */
783 MLD_CT_TESTING_DECLASSIFY(challenge_bytes, MLDSA_CTILDEBYTES);
784 /* @[FIPS204, Algorithm 7, line 16] c <- SampleInBall(ctilde) and
785 * @[FIPS204, Algorithm 7, line 17] c_hat <- NTT(c). */
786 mld_poly_challenge(cp, challenge_bytes);
787 mld_poly_ntt(cp);
788
789 /* @[FIPS204, Algorithm 7, lines 18+20] cs1 <- invNTT(c_hat o s1_hat) and
790 * z <- y + cs1, followed by the line-23 norm check ||z||_inf >= GAMMA1 -
791 * BETA. mld_compute_pack_z fuses all three per polynomial and, on success,
792 * packs z into sig; it returns MLD_ERR_FAIL if the norm check rejects z. */
793 ret = mld_compute_pack_z(sig, cp, s1hat, y, t, z);
794 if (ret != 0)
795 {
796 goto cleanup;
797 }
798
799 /* The remaining steps realize @[FIPS204, Algorithm 7, lines 21-28] (the
800 * low-bits norm check and the hint h) via the faster alternative formulation
801 * of @[Round3_Spec, Section 5.1]. @[FIPS204] explicitly permits this: the
802 * note accompanying Algorithm 7 states that the validity checks on z and the
803 * computation of h may instead be implemented "as described in Section 5.1 of
804 * [6]", and that reference is @[Round3_Spec, Section 5.1].
805 *
806 * The loop below builds w0 - cs2 + ct0 in place in w0; w1 is unmodified, and
807 * is HighBits(w) from line 13. Those are the inputs to the streamlined
808 * computation of MakeHint explained below.
809 *
810 * Low-bits norm check:
811 * @[FIPS204, Algorithm 7, line 21] computes r0 = LowBits(w - cs2) and line
812 * 23 rejects when ||r0||_inf >= GAMMA2 - BETA. By @[Round3_Spec, Section
813 * 5.1] (Lemma 3), this line-23 check on r0 = LowBits(w - cs2) is implied by
814 * ||w0 - cs2||_inf < GAMMA2 - BETA, where w0 is the low part of w. In our
815 * context, w0 already holds the low part of w from the line-13 Decompose;
816 * after subtracting cs2 from it in place, the mld_poly_chknorm(w0, GAMMA2 -
817 * BETA) call below is exactly that check.
818 *
819 * Hint:
820 * @[FIPS204, Algorithm 7, line 26] sets h = MakeHint(-ct0, w - cs2 + ct0),
821 * and line 28 rejects when ||ct0||_inf >= GAMMA2 or h has more than OMEGA
822 * nonzero coefficients. @[Round3_Spec, Section 5.1] provides the following
823 * alternative description for MakeHint(-ct0, w - cs2 + ct0): a hint bit is
824 * zero exactly when the coefficient of w0 - cs2 + ct0 lies in
825 * (-GAMMA2, GAMMA2], or equals -GAMMA2 while the matching w1 coefficient is
826 * zero (the Decompose border case), and is set otherwise. This equivalence
827 * is precisely what mld_pack_sig_h -> mld_make_hint compute from w0
828 * (= w0 - cs2 + ct0) and w1. The line-28 ||ct0||_inf >= GAMMA2 check is the
829 * mld_poly_chknorm(z, GAMMA2) call on ct0 below; the weight bound is
830 * enforced by mld_pack_sig_h.
831 *
832 * Building w0 per-component and checking norms incrementally also avoids
833 * allocating a full polyveck for h. */
834 for (k = 0; k < MLDSA_K; k++)
835 __loop__(
836 assigns(k,
837 object_whole(z),
838 object_whole(w0))
839 invariant(k <= MLDSA_K)
840 invariant(forall(k0, k, MLDSA_K,
841 array_abs_bound(w0->vec[k0].coeffs, 0, MLDSA_N, MLDSA_GAMMA2 + 1)))
842 decreases(MLDSA_K - k)
843 )
844 {
845 /* @[FIPS204, Algorithm 7, line 19] cs2[k] <- invNTT(c_hat o s2_hat)[k],
846 * then subtract from w0[k] to form (w0 - cs2)[k]. */
847 mld_sk_s2hat_get_poly(z, s2hat, k);
848 mld_poly_pointwise_montgomery(z, cp);
849 mld_poly_invntt_tomont(z);
850
851 mld_poly_sub(&w0->vec[k], z);
852 mld_poly_reduce(&w0->vec[k]);
853
854 /* Low-bits norm check (see block comment above): the line-23 check on
855 * r0 = LowBits(w - cs2) holds via ||w0 - cs2||_inf < GAMMA2 - BETA. */
856 w0_invalid = mld_poly_chknorm(&w0->vec[k], MLDSA_GAMMA2 - MLDSA_BETA);
857 /* Constant time: w0_invalid may be leaked - see comment for z_invalid. */
858 MLD_CT_TESTING_DECLASSIFY(&w0_invalid, sizeof(uint32_t));
859 if (w0_invalid)
860 {
861 ret = MLD_ERR_FAIL; /* reject */
862 goto cleanup;
863 }
864
865 /* @[FIPS204, Algorithm 7, line 25] ct0[k] <- invNTT(c_hat o t0_hat)[k]. */
866 mld_sk_t0hat_get_poly(z, t0hat, k);
867 mld_poly_pointwise_montgomery(z, cp);
868 mld_poly_invntt_tomont(z);
869 mld_poly_reduce(z);
870
871 /* @[FIPS204, Algorithm 7, line 28] reject when ||ct0||_inf >= GAMMA2 (the
872 * second part, the OMEGA weight bound, is enforced by mld_pack_sig_h). */
873 h_invalid = mld_poly_chknorm(z, MLDSA_GAMMA2);
874 /* Constant time: h_invalid may be leaked - see comment for z_invalid. */
875 MLD_CT_TESTING_DECLASSIFY(&h_invalid, sizeof(uint32_t));
876 if (h_invalid)
877 {
878 ret = MLD_ERR_FAIL; /* reject */
879 goto cleanup;
880 }
881
882 /* Add ct0[k] to (w0 - cs2)[k], leaving (w0 - cs2 + ct0)[k] in w0[k] -- the
883 * MakeHint input prepared for mld_pack_sig_h (see block comment above). */
884 mld_poly_add(&w0->vec[k], z);
885 }
886
887 /* Constant time: At this point all norm checks have passed and we, hence,
888 * know that the signature does not leak any secret information.
889 * Consequently, any value that can be computed from the signature and public
890 * key is considered public.
891 * w0 and w1 are public as they can be computed from Az - ct = \alpha w1 + w0.
892 * h=c*t0 is public as both c and t0 are considered public.
893 * While t0 is not part of the public key, it can be reconstructed from
894 * a small number of signatures and need not be regarded as secret
895 * (see @[FIPS204, Section 6.1]).
896 */
897 MLD_CT_TESTING_DECLASSIFY(w0, sizeof(*w0));
898 MLD_CT_TESTING_DECLASSIFY(w1, sizeof(*w1));
899
900 /* @[FIPS204, Algorithm 7, line 33] sigEncode(ctilde, z mod+/- q, h) is split
901 * across three calls: z was already packed by mld_compute_pack_z, this call
902 * packs ctilde, and mld_pack_sig_h below packs the hint h. */
903 mld_pack_sig_c(sig, challenge_bytes);
904
905 /* @[FIPS204, Algorithm 7, line 26] h <- MakeHint(-ct0, w - cs2 + ct0),
906 * computed from (w0 = w0 - cs2 + ct0, w1) as described in the block comment
907 * above, and packed as the h component of the line-33 sigEncode. Returns
908 * MLD_ERR_FAIL if h would exceed OMEGA nonzero coefficients (the remaining
909 * part of the line-28 check), in which case we reject. */
910 ret = mld_pack_sig_h(sig, w0, w1);
911 if (ret != 0)
912 {
913 goto cleanup;
914 }
915
916 /* Constant time: At this point it is clear that the signature is valid - it
917 * can, hence, be considered public. */
918 MLD_CT_TESTING_DECLASSIFY(sig, MLDSA_CRYPTO_BYTES);
919 ret = 0; /* success */
920
921cleanup:
922 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
923 MLD_FREE(t, mld_poly, 1, context);
924 MLD_FREE(cp, mld_poly, 1, context);
925 MLD_FREE(w0, mld_polyveck, 1, context);
926 MLD_FREE(w1tmp, w1tmp_u, 1, context);
927 MLD_FREE(z, mld_poly, 1, context);
928 MLD_FREE(y, mld_yvec, 1, context);
929 MLD_FREE(challenge_bytes, uint8_t, MLDSA_CTILDEBYTES, context);
930
931 return ret;
932}

934MLD_EXTERNAL_API
935int mld_sign_signature_internal(uint8_t sig[MLDSA_CRYPTO_BYTES],
936 const uint8_t *m, size_t mlen,
937 const uint8_t *pre, size_t prelen,
938 const uint8_t rnd[MLDSA_RNDBYTES],
939 const uint8_t sk[MLDSA_CRYPTO_SECRETKEYBYTES],
940 int externalmu,
941 MLD_CONFIG_CONTEXT_PARAMETER_TYPE context)
942{
943 int ret;
944 uint8_t *rho, *tr, *key, *mu, *rhoprime;
945 uint16_t attempt;
946 const uint16_t max_signing_attempts = mld_get_max_signing_attempts();
947 MLD_ALLOC(seedbuf, uint8_t,
948 2 * MLDSA_SEEDBYTES + MLDSA_TRBYTES + 2 * MLDSA_CRHBYTES, context);
949 MLD_ALLOC(mat, mld_polymat, 1, context);
950 MLD_ALLOC(s1hat, mld_sk_s1hat, 1, context);
951 MLD_ALLOC(t0hat, mld_sk_t0hat, 1, context);
952 MLD_ALLOC(s2hat, mld_sk_s2hat, 1, context);
953
954 if (seedbuf == NULL || mat == NULL || s1hat == NULL || t0hat == NULL ||
955 s2hat == NULL)
956 {
957 ret = MLD_ERR_OUT_OF_MEMORY;
958 goto cleanup;
959 }
960
961 /* If a resume hook is configured via MLD_CONFIG_SIGN_HOOK_RESUME, it provides
962 * the attempt to resume from after an earlier pause. Otherwise, we start at
963 * 0. Clamp to max_signing_attempts. */
964 attempt = mld_sign_resume(context);
965 if (attempt > max_signing_attempts)
966 {
967 attempt = max_signing_attempts;
968 }
969
970 rho = seedbuf;
971 tr = rho + MLDSA_SEEDBYTES;
972 key = tr + MLDSA_TRBYTES;
973 mu = key + MLDSA_SEEDBYTES;
974 rhoprime = mu + MLDSA_CRHBYTES;
975 /* @[FIPS204, Algorithm 7, line 1] (rho, K, tr, s1, s2, t0) <- skDecode(sk)
976 * and @[FIPS204, Algorithm 7, lines 2-4] s1_hat/s2_hat/t0_hat <- NTT(...):
977 * mld_unpack_sk returns s1hat, s2hat, t0hat already in NTT domain. The spec's
978 * private random seed K is held in the local variable key. */
979 mld_unpack_sk(rho, tr, key, t0hat, s1hat, s2hat, sk);
980
981 if (!externalmu)
982 {
983 /* @[FIPS204, Algorithm 7, line 6] mu <- H(BytesToBits(tr) || M', 64). */
984 mld_H(mu, MLDSA_CRHBYTES, tr, MLDSA_TRBYTES, pre, prelen, m, mlen);
985 }
986 else
987 {
988 /* mu has been provided directly (external-mu variant; line 6 done by the
989 * caller in a separate cryptographic module). */
990 mld_memcpy(mu, m, MLDSA_CRHBYTES);
991 }
992
993 /* @[FIPS204, Algorithm 7, line 7] rhoprime <- H(K || rnd || mu, 64). */
994 mld_H(rhoprime, MLDSA_CRHBYTES, key, MLDSA_SEEDBYTES, rnd, MLDSA_RNDBYTES, mu,
995 MLDSA_CRHBYTES);
996
997 /* Constant time: rho is part of the public key and, hence, public. */
998 MLD_CT_TESTING_DECLASSIFY(rho, MLDSA_SEEDBYTES);
999 /* @[FIPS204, Algorithm 7, line 5] A_hat <- ExpandA(rho). */
1000 mld_polyvec_matrix_expand(mat, rho);
1001
1002 /* @[FIPS204, Algorithm 7, lines 8-10 and 31-32] the rejection-sampling loop,
1003 * tracked by attempt (kappa = attempt*MLDSA_L). Each iteration's body (lines
1004 * 11-30) plus, on success, the line-33 sigEncode are performed by
1005 * mld_attempt_signature_generation. */
1006
1007 /* Reference: the reference loops unboundedly; we instead iterate over the
1008 * bounded range [0, max_signing_attempts) for predictable termination.
1009 * A success or fatal error exits via goto cleanup; running to completion
1010 * means every attempt was rejected; with a FIPS compliant choice of
1011 * MLD_CONFIG_MAX_SIGNING_ATTEMPTS, this should never happen. */
1012 for (; attempt < max_signing_attempts; attempt++)
1013 __loop__(
1014 MLD_IF_NOT_REDUCE_RAM(
1015 assigns(attempt, ret, memory_slice(sig, MLDSA_CRYPTO_BYTES))
1016 )
1017 MLD_IF_REDUCE_RAM(
1018 assigns(attempt, ret, memory_slice(sig, MLDSA_CRYPTO_BYTES),
1019 memory_slice(mat, sizeof(mld_polymat)))
1020 )
1021 invariant(attempt <= max_signing_attempts)
1022
1023 /* t0, s1, s2, and mat are initialized above and are NOT changed by this */
1024 /* loop. We can therefore re-assert their bounds here as part of the */
1025 /* loop invariant. This makes proof noticeably faster with CBMC */
1026 MLD_IF_NOT_REDUCE_RAM(
1027 invariant(forall(k1, 0, MLDSA_K, forall(l1, 0, MLDSA_L,
1028 array_bound(mat->vec[k1].vec[l1].coeffs, 0, MLDSA_N, 0, MLDSA_Q))))
1029 invariant(forall(k2, 0, MLDSA_K, array_abs_bound(t0hat->vec.vec[k2].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
1030 invariant(forall(k3, 0, MLDSA_L, array_abs_bound(s1hat->vec.vec[k3].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
1031 invariant(forall(k4, 0, MLDSA_K, array_abs_bound(s2hat->vec.vec[k4].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
1032 )
1033 decreases(max_signing_attempts - attempt)
1034 )
1035 {
1036 /* Safety: attempt < max_signing_attempts <= MLD_MAX_KAPPA / MLDSA_L, so
1037 * kappa <= MLD_MAX_KAPPA and the cast is safe. */
1038 const uint16_t kappa = (uint16_t)(attempt * MLDSA_L);
1039
1040 /* Query configurable signing hook whether signing should be paused.
1041 * This is skipped by default and only used if the user sets the
1042 * configuration option MLD_CONFIG_SIGN_HOOK_ATTEMPT. */
1043 if (mld_sign_attempt(attempt, context) != 0)
1044 {
1045 ret = MLD_ERR_SIGNING_PAUSED;
1046 goto cleanup;
1047 }
1048
1049 ret = mld_attempt_signature_generation(sig, mu, rhoprime, kappa, mat, s1hat,
1050 s2hat, t0hat, context);
1051
1052 /* Decide whether to keep trying based on the return value:
1053 * - ret == 0: a valid signature was produced; we are done.
1054 * - ret == MLD_ERR_FAIL: this candidate was rejected by one of the norm
1055 * or hint checks. We continue the loop and try again with the next
1056 * nonce.
1057 * - any other value (e.g. MLD_ERR_OUT_OF_MEMORY): an unrecoverable error
1058 * occurred, so we propagate it to the caller. */
1059 if (ret == 0)
1060 {
1061 /* Signing succeeded: record the attempt that succeeded. No-op in the
1062 * default build. */
1063 mld_sign_finish(attempt, context);
1064 goto cleanup;
1065 }
1066 if (ret != MLD_ERR_FAIL)
1067 {
1068 goto cleanup;
1069 }
1070 }
1071
1072 /* Loop ran to completion: all attempts rejected, budget exhausted.
1073 * This should never happen with a FIPS compliant choice of
1074 * MLD_CONFIG_MAX_SIGNING_ATTEMPTS. */
1075 ret = MLD_ERR_SIGN_ATTEMPTS_EXHAUSTED;
1076
1077cleanup:
1078
1079 if (ret != 0)
1080 {
1081 /* To be on the safe-side, we zeroize the signature buffer. */
1082 mld_zeroize(sig, MLDSA_CRYPTO_BYTES);
1083 }
1084
1085 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
1086 MLD_FREE(s2hat, mld_sk_s2hat, 1, context);
1087 MLD_FREE(t0hat, mld_sk_t0hat, 1, context);
1088 MLD_FREE(s1hat, mld_sk_s1hat, 1, context);
1089 MLD_FREE(mat, mld_polymat, 1, context);
1090 MLD_FREE(seedbuf, uint8_t,
1091 2 * MLDSA_SEEDBYTES + MLDSA_TRBYTES + 2 * MLDSA_CRHBYTES, context);
1092 return ret;
1093}

1195#if !defined(MLD_CONFIG_NO_VERIFY_API)
1196MLD_MUST_CHECK_RETURN_VALUE
1197MLD_EXTERNAL_API
1198int mld_sign_verify_internal(const uint8_t sig[MLDSA_CRYPTO_BYTES],
1199 const uint8_t *m, size_t mlen, const uint8_t *pre,
1200 size_t prelen,
1201 const uint8_t pk[MLDSA_CRYPTO_PUBLICKEYBYTES],
1202 int externalmu,
1203 MLD_CONFIG_CONTEXT_PARAMETER_TYPE context)
1204{
1205 int ret, cmp;
1206 unsigned int i;
1207
1208 MLD_ALLOC(buf, uint8_t, (MLDSA_K * MLDSA_POLYW1_PACKEDBYTES), context);
1209 MLD_ALLOC(mu, uint8_t, MLDSA_CRHBYTES, context);
1210 MLD_ALLOC(c, uint8_t, MLDSA_CTILDEBYTES, context);
1211 MLD_ALLOC(c2, uint8_t, MLDSA_CTILDEBYTES, context);
1212 MLD_ALLOC(z, mld_polyvecl, 1, context);
1213 MLD_ALLOC(cp, mld_poly, 1, context);
1214 MLD_ALLOC(mat, mld_polymat, 1, context);
1215 MLD_ALLOC(w1, mld_poly, 1, context);
1216 MLD_ALLOC(tmp, mld_poly, 1, context);
1217
1218 if (buf == NULL || mu == NULL || c == NULL || c2 == NULL || z == NULL ||
1219 cp == NULL || mat == NULL || w1 == NULL || tmp == NULL)
1220 {
1221 ret = MLD_ERR_OUT_OF_MEMORY;
1222 goto cleanup;
1223 }
1224
1225 mld_memcpy(c, sig, MLDSA_CTILDEBYTES);
1226 mld_polyvecl_unpack_z(z, sig + MLDSA_SIG_Z_OFFSET);
1227
1228 /* mld_polyvecl_chknorm signals failure through a single non-zero error code
1229 * that's not yet aligned with MLD_ERR_XXX. A norm-check failure here means
1230 * the signature is invalid, so map it to MLD_ERR_INVALID_SIGNATURE. */
1231 if (mld_polyvecl_chknorm(z, MLDSA_GAMMA1 - MLDSA_BETA))
1232 {
1233 ret = MLD_ERR_INVALID_SIGNATURE;
1234 goto cleanup;
1235 }
1236
1237 if (!externalmu)
1238 {
1239 /* Compute CRH(H(rho, t1), pre, msg) */
1240 MLD_ALIGN uint8_t hpk[MLDSA_CRHBYTES];
1241 mld_H(hpk, MLDSA_TRBYTES, pk, MLDSA_CRYPTO_PUBLICKEYBYTES, NULL, 0, NULL,
1242 0);
1243 mld_H(mu, MLDSA_CRHBYTES, hpk, MLDSA_TRBYTES, pre, prelen, m, mlen);
1244
1245 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
1246 mld_zeroize(hpk, sizeof(hpk));
1247 }
1248 else
1249 {
1250 /* mu has been provided directly */
1251 mld_memcpy(mu, m, MLDSA_CRHBYTES);
1252 }
1253
1254 /* Matrix-vector multiplication and per-row reconstruction of w1. */
1255 mld_polyvecl_ntt(z);
1256 mld_polyvec_matrix_expand(mat, pk);
1257 mld_poly_challenge(cp, c);
1258 mld_poly_ntt(cp);
1259
1260 for (i = 0; i < MLDSA_K; ++i)
1261 __loop__(
1262 assigns(MLD_IF_REDUCE_RAM(memory_slice(mat, sizeof(mld_polymat)),)
1263 i, ret,
1264 memory_slice(w1, sizeof(mld_poly)),
1265 memory_slice(tmp, sizeof(mld_poly)),
1266 memory_slice(buf, MLDSA_K * MLDSA_POLYW1_PACKEDBYTES)
1267 )
1268 invariant(i <= MLDSA_K)
1269 decreases(MLDSA_K - i)
1270 )
1271 {
1272 /* w1 = (A * z)_i in NTT domain */
1273 mld_polyvec_matrix_pointwise_montgomery_row(w1, mat, z, i);
1274
1275 /* tmp = c * t1_i * 2^d in NTT domain */
1276 mld_unpack_pk_t1(tmp, pk, i);
1277 mld_poly_shiftl(tmp);
1278 mld_poly_ntt(tmp);
1279 mld_poly_pointwise_montgomery(tmp, cp);
1280
1281 /* w1 = invNTT(w1 - c * t1_i * 2^d) */
1282 mld_poly_sub(w1, tmp);
1283 mld_poly_reduce(w1);
1284 mld_poly_invntt_tomont(w1);
1285 mld_poly_caddq(w1);
1286
1287 /* tmp = h_i (decoded and validated from signature). A non-zero return
1288 * means the hint encoding is malformed, i.e. the signature is invalid. */
1289 if (mld_sig_unpack_hints(tmp, sig, i) != 0)
1290 {
1291 ret = MLD_ERR_INVALID_SIGNATURE;
1292 goto cleanup;
1293 }
1294
1295 /* w1 = use_hint(w1, tmp), then pack into buf[i] */
1296 mld_poly_use_hint(w1, tmp);
1297 mld_polyw1_pack(buf + i * MLDSA_POLYW1_PACKEDBYTES, w1);
1298 }
1299
1300 /* Call random oracle and verify challenge */
1301 mld_H(c2, MLDSA_CTILDEBYTES, mu, MLDSA_CRHBYTES, buf,
1302 MLDSA_K * MLDSA_POLYW1_PACKEDBYTES, NULL, 0);
1303
1304 cmp = mld_ct_memcmp(c, c2, MLDSA_CTILDEBYTES);
1305
1306 /* Declassify the result of the verification. */
1307 MLD_CT_TESTING_DECLASSIFY(&cmp, sizeof(cmp));
1308
1309 ret = cmp == 0 ? 0 : MLD_ERR_INVALID_SIGNATURE;
1310
1311cleanup:
1312 /* @[FIPS204, Section 3.6.3] Destruction of intermediate values. */
1313 MLD_FREE(tmp, mld_poly, 1, context);
1314 MLD_FREE(w1, mld_poly, 1, context);
1315 MLD_FREE(mat, mld_polymat, 1, context);
1316 MLD_FREE(cp, mld_poly, 1, context);
1317 MLD_FREE(z, mld_polyvecl, 1, context);
1318 MLD_FREE(c2, uint8_t, MLDSA_CTILDEBYTES, context);
1319 MLD_FREE(c, uint8_t, MLDSA_CTILDEBYTES, context);
1320 MLD_FREE(mu, uint8_t, MLDSA_CRHBYTES, context);
1321 MLD_FREE(buf, uint8_t, (MLDSA_K * MLDSA_POLYW1_PACKEDBYTES), context);
1322 return ret;
1323}

1581MLD_EXTERNAL_API
1582size_t mld_prepare_domain_separation_prefix(
1583 uint8_t prefix[MLD_DOMAIN_SEPARATION_MAX_BYTES], const uint8_t *ph,
1584 size_t phlen, const uint8_t *ctx, size_t ctxlen, int hashalg)
1585{
1586 if (ctxlen > 255)
1587 {
1588 return 0;
1589 }
1590
1591 if (hashalg != MLD_PREHASH_NONE)
1592 {
1593 if (ph == NULL || mld_validate_hash_length(hashalg, phlen) != 0)
1594 {
1595 return 0;
1596 }
1597 }
1598
1599 /* Common prefix: 0x00/0x01 || ctxlen || ctx */
1600 prefix[0] = (hashalg == MLD_PREHASH_NONE) ? 0 : 1;
1601 prefix[1] = (uint8_t)ctxlen;
1602 if (ctxlen > 0)
1603 {
1604 mld_memcpy(prefix + 2, ctx, ctxlen);
1605 }
1606
1607 if (hashalg == MLD_PREHASH_NONE)
1608 {
1609 return 2 + ctxlen;
1610 }
1611
1612 /* HashML-DSA: append oid || ph */
1613 mld_get_hash_oid(prefix + 2 + ctxlen, hashalg);
1614 mld_memcpy(prefix + 2 + ctxlen + MLD_PRE_HASH_OID_LEN, ph, phlen);
