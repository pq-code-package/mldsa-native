28#if !defined(MLD_CONFIG_NO_SIGN_API) && \
29 (!defined(MLD_CONFIG_REDUCE_RAM) || defined(MLD_UNIT_TEST))
30MLD_INTERNAL_API
31void mld_polyvecl_uniform_gamma1(mld_polyvecl *v,
32 const uint8_t seed[MLDSA_CRHBYTES],
33 uint16_t kappa)
34{
35#if defined(MLD_CONFIG_SERIAL_FIPS202_ONLY)
36 int i;
37#endif
38
39 /* The caller passes the base counter kappa; component i is sampled from
40 * kappa + i. Safety: kappa <= MLD_MAX_KAPPA and i < MLDSA_L, so the
41 * casts below are safe. See MLD_MAX_KAPPA comment in params.h. */
42#if defined(MLD_CONFIG_SERIAL_FIPS202_ONLY)
43 for (i = 0; i < MLDSA_L; i++)
44 {
45 mld_poly_uniform_gamma1(&v->vec[i], seed, (uint16_t)(kappa + i));
46 }
47#else /* MLD_CONFIG_SERIAL_FIPS202_ONLY */
48#if MLDSA_L == 4
49 mld_poly_uniform_gamma1_4x(&v->vec[0], &v->vec[1], &v->vec[2], &v->vec[3],
50 seed, kappa, (uint16_t)(kappa + 1),
51 (uint16_t)(kappa + 2), (uint16_t)(kappa + 3));
52#elif MLDSA_L == 5
53 mld_poly_uniform_gamma1_4x(&v->vec[0], &v->vec[1], &v->vec[2], &v->vec[3],
54 seed, kappa, (uint16_t)(kappa + 1),
55 (uint16_t)(kappa + 2), (uint16_t)(kappa + 3));
56 mld_poly_uniform_gamma1(&v->vec[4], seed, (uint16_t)(kappa + 4));
57#elif MLDSA_L == 7
58 mld_poly_uniform_gamma1_4x(&v->vec[0], &v->vec[1], &v->vec[2],
59 &v->vec[3 /* irrelevant */], seed, kappa,
60 (uint16_t)(kappa + 1), (uint16_t)(kappa + 2),
61 0xFF /* irrelevant */);
62 mld_poly_uniform_gamma1_4x(&v->vec[3], &v->vec[4], &v->vec[5], &v->vec[6],
63 seed, (uint16_t)(kappa + 3), (uint16_t)(kappa + 4),
64 (uint16_t)(kappa + 5), (uint16_t)(kappa + 6));
65#endif /* MLDSA_L == 7 */
66#endif /* !MLD_CONFIG_SERIAL_FIPS202_ONLY */
67
68 mld_assert_bound_2d(v->vec, MLDSA_L, MLDSA_N, -(MLDSA_GAMMA1 - 1),
69 MLDSA_GAMMA1 + 1);
70}
71#endif /* !MLD_CONFIG_NO_SIGN_API && (!MLD_CONFIG_REDUCE_RAM || MLD_UNIT_TEST) \
72 */

101#if !defined(MLD_CONFIG_REDUCE_RAM) || defined(MLD_UNIT_TEST)
102MLD_STATIC_TESTABLE void mld_polyvecl_pointwise_acc_montgomery_c(
103 mld_poly *w, const mld_polyvecl *u, const mld_polyvecl *v)
104__contract__(
105 requires(memory_no_alias(w, sizeof(mld_poly)))
106 requires(memory_no_alias(u, sizeof(mld_polyvecl)))
107 requires(memory_no_alias(v, sizeof(mld_polyvecl)))
108 requires(forall(l0, 0, MLDSA_L,
109 array_bound(u->vec[l0].coeffs, 0, MLDSA_N, 0, MLDSA_Q)))
110 requires(forall(l1, 0, MLDSA_L,
111 array_abs_bound(v->vec[l1].coeffs, 0, MLDSA_N, MLD_NTT_BOUND)))
112 assigns(memory_slice(w, sizeof(mld_poly)))
113 ensures(array_abs_bound(w->coeffs, 0, MLDSA_N, MLDSA_Q))
114)
115{
116 unsigned int i, j;
117 mld_assert_bound_2d(u->vec, MLDSA_L, MLDSA_N, 0, MLDSA_Q);
118 mld_assert_abs_bound_2d(v->vec, MLDSA_L, MLDSA_N, MLD_NTT_BOUND);
119 for (i = 0; i < MLDSA_N; i++)
120 __loop__(
121 assigns(i, j, memory_slice(w, sizeof(mld_poly)))
122 invariant(i <= MLDSA_N)
123 invariant(array_abs_bound(w->coeffs, 0, i, MLDSA_Q))
124 decreases(MLDSA_N - i)
125 )
126 {
127 int64_t t = 0;
128 int32_t r;
129 for (j = 0; j < MLDSA_L; j++)
130 __loop__(
131 assigns(j, t)
132 invariant(j <= MLDSA_L)
133 invariant(t >= -(int64_t)j*(MLDSA_Q - 1)*(MLD_NTT_BOUND - 1))
134 invariant(t <= (int64_t)j*(MLDSA_Q - 1)*(MLD_NTT_BOUND - 1))
135 decreases(MLDSA_L - j)
136 )
137 {
138 t += (int64_t)u->vec[j].coeffs[i] * v->vec[j].coeffs[i];
139 }
140
141 r = mld_montgomery_reduce(t);
142 w->coeffs[i] = r;
143 }
144
145 mld_assert_abs_bound(w->coeffs, MLDSA_N, MLDSA_Q);
146}

148MLD_INTERNAL_API
149void mld_polyvecl_pointwise_acc_montgomery(mld_poly *w, const mld_polyvecl *u,
150 const mld_polyvecl *v)
151{
152#if defined(MLD_USE_NATIVE_POLYVECL_POINTWISE_ACC_MONTGOMERY_L4) && \
153 MLD_CONFIG_PARAMETER_SET == 44
154 int ret;
155 mld_assert_bound_2d(u->vec, MLDSA_L, MLDSA_N, 0, MLDSA_Q);
156 mld_assert_abs_bound_2d(v->vec, MLDSA_L, MLDSA_N, MLD_NTT_BOUND);
157 ret = mld_polyvecl_pointwise_acc_montgomery_l4_native(
158 w->coeffs, (const int32_t (*)[MLDSA_N])u->vec,
159 (const int32_t (*)[MLDSA_N])v->vec);
160 if (ret == MLD_NATIVE_FUNC_SUCCESS)
161 {
162 mld_assert_abs_bound(w->coeffs, MLDSA_N, MLDSA_Q);
163 return;
164 }
165#elif defined(MLD_USE_NATIVE_POLYVECL_POINTWISE_ACC_MONTGOMERY_L5) && \
166 MLD_CONFIG_PARAMETER_SET == 65
167 int ret;
168 mld_assert_bound_2d(u->vec, MLDSA_L, MLDSA_N, 0, MLDSA_Q);
169 mld_assert_abs_bound_2d(v->vec, MLDSA_L, MLDSA_N, MLD_NTT_BOUND);
170 ret = mld_polyvecl_pointwise_acc_montgomery_l5_native(
171 w->coeffs, (const int32_t (*)[MLDSA_N])u->vec,
172 (const int32_t (*)[MLDSA_N])v->vec);
173 if (ret == MLD_NATIVE_FUNC_SUCCESS)
174 {
175 mld_assert_abs_bound(w->coeffs, MLDSA_N, MLDSA_Q);
176 return;
177 }
178#elif defined(MLD_USE_NATIVE_POLYVECL_POINTWISE_ACC_MONTGOMERY_L7) && \
179 MLD_CONFIG_PARAMETER_SET == 87
180 int ret;
181 mld_assert_bound_2d(u->vec, MLDSA_L, MLDSA_N, 0, MLDSA_Q);
182 mld_assert_abs_bound_2d(v->vec, MLDSA_L, MLDSA_N, MLD_NTT_BOUND);
183 ret = mld_polyvecl_pointwise_acc_montgomery_l7_native(
184 w->coeffs, (const int32_t (*)[MLDSA_N])u->vec,
185 (const int32_t (*)[MLDSA_N])v->vec);
186 if (ret == MLD_NATIVE_FUNC_SUCCESS)
187 {
188 mld_assert_abs_bound(w->coeffs, MLDSA_N, MLDSA_Q);
189 return;
190 }
191#endif /* !(MLD_USE_NATIVE_POLYVECL_POINTWISE_ACC_MONTGOMERY_L4 && \
192 MLD_CONFIG_PARAMETER_SET == 44) && \
193 !(MLD_USE_NATIVE_POLYVECL_POINTWISE_ACC_MONTGOMERY_L5 && \
194 MLD_CONFIG_PARAMETER_SET == 65) && \
195 MLD_USE_NATIVE_POLYVECL_POINTWISE_ACC_MONTGOMERY_L7 && \
196 MLD_CONFIG_PARAMETER_SET == 87 */
197 /* The first input is bounded by [0, MLDSA_Q-1] inclusive.
198 * The second input is bounded by [-(MLD_NTT_BOUND-1), MLD_NTT_BOUND-1].
199 * Hence, we can safely accumulate in 64-bits without intermediate reductions
200 * as MLDSA_L * (MLD_NTT_BOUND-1) * (MLDSA_Q-1) < INT64_MAX.
201 *
202 * The worst case is ML-DSA-87: 7 * (MLD_NTT_BOUND-1) * (MLDSA_Q-1) < 2**53
203 * (and likewise for negative values).
204 */
205 mld_polyvecl_pointwise_acc_montgomery_c(w, u, v);
206}
207#endif /* !MLD_CONFIG_REDUCE_RAM || MLD_UNIT_TEST */

209#if !defined(MLD_CONFIG_NO_KEYPAIR_API) || !defined(MLD_CONFIG_NO_VERIFY_API) || \
210 defined(MLD_UNIT_TEST)
211MLD_INTERNAL_API
212uint32_t mld_polyvecl_chknorm(const mld_polyvecl *v, int32_t bound)
213{
214 unsigned int i;
215 uint32_t t = 0;
216 mld_assert_bound_2d(v->vec, MLDSA_L, MLDSA_N, -MLD_REDUCE32_RANGE_MAX,
217 MLD_REDUCE32_RANGE_MAX);
218
219 for (i = 0; i < MLDSA_L; ++i)
220 __loop__(
221 invariant(i <= MLDSA_L)
222 invariant(t == 0 || t == 0xFFFFFFFF)
223 invariant((t == 0) == forall(k1, 0, i, array_abs_bound(v->vec[k1].coeffs, 0, MLDSA_N, bound)))
224 decreases(MLDSA_L - i)
225 )
226 {
227 /* Reference: Leaks which polynomial violates the bound via a conditional.
228 * We are more conservative to reduce the number of declassifications in
229 * constant-time testing.
230 */
231 t |= mld_poly_chknorm(&v->vec[i], bound);
232 }
233 return t;
234}
235#endif /* !MLD_CONFIG_NO_KEYPAIR_API || !MLD_CONFIG_NO_VERIFY_API || \

367MLD_INTERNAL_API
368void mld_polyveck_decompose(mld_polyveck *v1, mld_polyveck *v0)
369{
370 unsigned int i;
371 mld_assert_bound_2d(v0->vec, MLDSA_K, MLDSA_N, 0, MLDSA_Q);
372
373 for (i = 0; i < MLDSA_K; ++i)
374 __loop__(
375 assigns(i, memory_slice(v0, sizeof(mld_polyveck)), memory_slice(v1, sizeof(mld_polyveck)))
376 invariant(i <= MLDSA_K)
377 invariant(forall(k1, 0, i,
378 array_bound(v1->vec[k1].coeffs, 0, MLDSA_N, 0, (MLDSA_Q-1)/(2*MLDSA_GAMMA2))))
379 invariant(forall(k2, 0, i,
380 array_abs_bound(v0->vec[k2].coeffs, 0, MLDSA_N, MLDSA_GAMMA2+1)))
381 invariant(forall(k3, i, MLDSA_K,
382 array_bound(v0->vec[k3].coeffs, 0, MLDSA_N, 0, MLDSA_Q)))
383 decreases(MLDSA_K - i)
384 )
385 {
386 mld_poly_decompose(&v1->vec[i], &v0->vec[i]);
387 }
388
389 mld_assert_bound_2d(v1->vec, MLDSA_K, MLDSA_N, 0,
390 (MLDSA_Q - 1) / (2 * MLDSA_GAMMA2));
391 mld_assert_abs_bound_2d(v0->vec, MLDSA_K, MLDSA_N, MLDSA_GAMMA2 + 1);
392}
