32/* Constant-time comparisons and conditional operations
33
34 We reduce the risk for compilation into variable-time code
35 through the use of 'value barriers'.
36
37 Functionally, a value barrier is a no-op. To the compiler, however,
38 it constitutes an arbitrary modification of its input, and therefore
39 harden's value propagation and range analysis.
40
41 We consider two approaches to implement a value barrier:
42 - An empty inline asm block which marks the target value as clobbered.
43 - XOR'ing with the value of a volatile global that's set to 0;
44 see @[optblocker] for a discussion of this idea, and
45 @[libmceliece, inttypes/crypto_intN.h] for an implementation.
46
47 The first approach is cheap because it only prevents the compiler
48 from reasoning about the value of the variable past the barrier,
49 but does not directly generate additional instructions.
50
51 The second approach generates redundant loads and XOR operations
52 and therefore comes at a higher runtime cost. However, it appears
53 more robust towards optimization, as compilers should never drop
54 a volatile load.
55
56 We use the empty-ASM value barrier for GCC and clang, and fall
57 back to the global volatile barrier otherwise.
58
59 The global value barrier can be forced by setting
60 MLD_CONFIG_NO_ASM_VALUE_BARRIER.
61
62*/

210MLD_MUST_CHECK_RETURN_VALUE
211static MLD_INLINE int32_t mld_ct_sel_int32(int32_t a, int32_t b, uint32_t cond)
212__contract__(
213 requires(cond == 0x0 || cond == 0xFFFFFFFF)
214 ensures(return_value == (cond ? a : b))
215)
216{
217 uint32_t au = mld_cast_int32_to_uint32(a);
218 uint32_t bu = mld_cast_int32_to_uint32(b);
219 uint32_t res = bu ^ (mld_value_barrier_u32(cond) & (au ^ bu));
220 return mld_cast_uint32_to_int32(res);
221}

228MLD_MUST_CHECK_RETURN_VALUE
229static MLD_INLINE uint32_t mld_ct_cmask_nonzero_u32(uint32_t x)
230__contract__(ensures(return_value == ((x == 0) ? 0 : 0xFFFFFFFF)))
231{
232 int64_t tmp = mld_value_barrier_i64(-((int64_t)x));
233 /*
234 * PORTABILITY: Right-shift on a signed integer is
235 * implementation-defined for negative left argument.
236 * Here, we assume it's sign-preserving "arithmetic" shift right.
237 * See (C99 6.5.7 (5))
238 */
239 tmp >>= 32;
240 return mld_cast_int64_to_uint32(tmp);
241}

303MLD_MUST_CHECK_RETURN_VALUE
304static MLD_INLINE uint8_t mld_ct_memcmp(const uint8_t *a, const uint8_t *b,
305 const size_t len)
306__contract__(
307 requires(len <= UINT16_MAX)
308 requires(memory_no_alias(a, len))
309 requires(memory_no_alias(b, len))
310 ensures((return_value == 0) || (return_value == 0xFF))
311 ensures((return_value == 0) == forall(i, 0, len, (a[i] == b[i]))))
312{
313 uint8_t r = 0, s = 0;
314 unsigned i;
315
316 for (i = 0; i < len; i++)
317 __loop__(
318 invariant(i <= len)
319 invariant((r == 0) == (forall(k, 0, i, (a[k] == b[k]))))
320 decreases(len - i))
321 {
322 r |= a[i] ^ b[i];
323 /* s is useless, but prevents the loop from being aborted once r=0xff. */
324 s ^= a[i] ^ b[i];
325 }
326
327 /*
328 * - Convert r into a mask; this may not be necessary, but is an additional
329 * safeguard
330 * towards leaking information about a and b.
331 * - XOR twice with s, separated by a value barrier, to prevent the compile
332 * from dropping the s computation in the loop.
333 */
334 return (mld_value_barrier_u8(mld_ct_cmask_nonzero_u8(r) ^ s) ^ s);
335}

352static MLD_INLINE void mld_zeroize(void *ptr, size_t len)
353__contract__(
354 requires(len <= UINT32_MAX)
355 requires(memory_no_alias(ptr, len))
356 assigns(memory_slice(ptr, len))
357 ensures(array_zeroized_u8((uint8_t *)ptr, len)))
358{
359#if defined(MLD_SYS_WINDOWS)
360 SecureZeroMemory(ptr, len);
361#else
362 mld_memset(ptr, 0, len);
363 /* This follows OpenSSL and seems sufficient to prevent the compiler
364 * from optimizing away the memset.
365 *
366 * If there was a reliable way to detect availability of memset_s(),
367 * that would be preferred. */
368 __asm__ __volatile__("" : : "r"(ptr) : "memory");
369#endif /* !MLD_SYS_WINDOWS */
370}
