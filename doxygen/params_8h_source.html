73/* Sampling y from counter kappa uses nonces kappa, ..., kappa+L-1, which fit in
74 * uint16_t iff kappa <= UINT16_MAX - MLDSA_L. With kappa = attempt*MLDSA_L this
75 * bounds the number of signing attempts by MLD_MAX_KAPPA / MLDSA_L; see
76 * MLD_MAX_SIGNING_ATTEMPTS in sign.c. */
77#define MLD_MAX_KAPPA (UINT16_MAX - MLDSA_L)
