103/*
104 * Quantifiers
105 * Note that the range on qvar is _exclusive_ between qvar_lb .. qvar_ub
106 * https://diffblue.github.io/cbmc/contracts-quantifiers.html
107 *
108 * The quantified variable is declared as uint32_t, so these macros
109 * quantify only over indices in [0, UINT32_MAX). Bounds larger than
110 * UINT32_MAX (4 GiB) are NOT supported: the explicit (uint32_t) casts
111 * on the bounds will trigger CBMC's conversion check if a wider bound
112 * (e.g. a size_t > UINT32_MAX) is passed.
113 *
114 * Quantifying over size_t (64-bit) was found to blow up SMT proof
115 * times, so we deliberately keep the index width at 32 bits. Callers
116 * dealing with size_t-typed buffers must add an explicit
117 * requires(len <= UINT32_MAX)
118 * precondition.
119 */
120

127 { \
128 uint32_t qvar; \
129 ((uint32_t) (qvar_lb) <= (qvar) && (qvar) < (uint32_t) (qvar_ub)) ==> (predicate) \
130 }

150#define array_bound_core(qvar, qvar_lb, qvar_ub, array_var, \
151 value_lb, value_ub) \
152 __CPROVER_forall \
153 { \
154 uint32_t qvar; \
155 ((uint32_t) (qvar_lb) <= (qvar) && (qvar) < (uint32_t) (qvar_ub)) ==> \
156 (((int)(value_lb) <= ((array_var)[(qvar)])) && \
157 (((array_var)[(qvar)]) < (int)(value_ub))) \
158 }

209/*
210 * Output-buffer discipline on failure, as documented in API-CONVENTIONS.md:
211 * when a function fails, each caller-owned output buffer is left either
212 * fully unchanged or fully zeroized -- never holding partially computed or
213 * stale data that could be mistaken for a valid result.
214 *
215 * Note the disjunction is over the buffer as a whole: it is not enough for
216 * each byte to be individually either unchanged or zero.
217 */
218#define array_unchanged_or_zeroized_u8(array_var, N) \
219 (array_unchanged_u8((array_var), (N)) || array_zeroized_u8((array_var), (N)))

221/* Wrapper around array_bound operating on absolute values.
222 *
223 * The absolute value bound `k` is exclusive.
224 *
225 * Note that since the lower bound in array_bound is inclusive, we have to
226 * raise it by 1 here.
227 */
228#define array_abs_bound(arr, lb, ub, k) \
229 array_bound((arr), (lb), (ub), -((int)(k)) + 1, (k))
