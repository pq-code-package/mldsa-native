33MLD_INTERNAL_API
34void mld_debug_check_bounds(const char *file, int line, const int32_t *ptr,
35 unsigned len, int64_t lower_bound_exclusive,
36 int64_t upper_bound_exclusive)
37{
38 int err = 0;
39 unsigned i;
40 for (i = 0; i < len; i++)
41 {
42 int32_t val = ptr[i];
43 if (!(val > lower_bound_exclusive && val < upper_bound_exclusive))
44 {
45 fprintf(stderr,
46 MLD_DEBUG_ERROR_HEADER
47 "Bounds assertion failed: Index %u, value %d out of bounds "
48 "(%" PRId64 ",%" PRId64 ")\n",
49 file, line, i, (int)val, lower_bound_exclusive,
50 upper_bound_exclusive);
51 err = 1;
52 }
53 }
54
55 if (err == 1)
56 {
57 exit(1);
58 }
59}
