28 * - On Arm-based Apple CPUs, or CPUs with MLD_SYS_AARCH64_FAST_SHA3 set,

30 * - Otherwise, unless MLD_SYS_AARCH64_SLOW_BARREL_SHIFTER is set,

32 * - Otherwise, if MLD_SYS_AARCH64_SLOW_BARREL_SHIFTER is set, we

48/*
49 * Keccak-f1600x2/x4
50 *
51 * The optimal implementation is highly CPU-specific; see @[HYBRID].
52 *
53 * For now, if v8.4-A is not implemented, we fall back to Keccak-f1600.
54 * If v8.4-A is implemented and we are on an Apple CPU (or a CPU with
55 * MLD_SYS_AARCH64_FAST_SHA3 set), we use a plain Neon-based implementation.
56 * If v8.4-A is implemented and we are on neither, we use a
57 * scalar/Neon/Neon hybrid.
58 * The reason for this distinction is that Apple CPUs (and other CPUs flagged
59 * via MLD_SYS_AARCH64_FAST_SHA3) appear to implement the SHA3 instructions on
60 * all SIMD units, while Arm CPUs prior to Cortex-X4 don't, and ordinary Neon
61 * instructions are still needed.
62 */
63#if defined(__ARM_FEATURE_SHA3)
64/*
65 * For Apple-M cores (and other cores flagged via MLD_SYS_AARCH64_FAST_SHA3),
66 * we use a plain implementation leveraging SHA3 instructions only.
67 */
68#if defined(__APPLE__) || defined(MLD_SYS_AARCH64_FAST_SHA3)
