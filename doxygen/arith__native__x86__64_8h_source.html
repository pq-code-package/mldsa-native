12#define MLD_AVX2_REJ_UNIFORM_BUFLEN \
13 (5 * 168) /* REJ_UNIFORM_NBLOCKS * SHAKE128_RATE */

16/*
17 * Sampling 256 coefficients mod 15 using rejection sampling from 4 bits.
18 * Expected number of required bytes: (256 * (16/15))/2 = 136.5 bytes.
19 * We sample 1 block (=136 bytes) of SHAKE256_RATE output initially.
20 * Sampling 2 blocks initially results in slightly worse performance.
21 */
22#define MLD_AVX2_REJ_UNIFORM_ETA2_BUFLEN (1 * 136)
23
24/*
25 * Sampling 256 coefficients mod 9 using rejection sampling from 4 bits.
26 * Expected number of required bytes: (256 * (16/9))/2 = 227.5 bytes.
27 * We sample 2 blocks (=272 bytes) of SHAKE256_RATE output initially.
28 */
29#define MLD_AVX2_REJ_UNIFORM_ETA4_BUFLEN (2 * 136)
