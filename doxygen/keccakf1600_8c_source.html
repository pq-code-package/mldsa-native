36MLD_INTERNAL_API
37void mld_keccakf1600_extract_bytes(uint64_t *state, unsigned char *data,
38 unsigned offset, unsigned length)
39{
40 unsigned i;
41#if defined(MLD_SYS_LITTLE_ENDIAN)
42 uint8_t *state_ptr = (uint8_t *)state + offset;
43 for (i = 0; i < length; i++)
44 __loop__(invariant(i <= length)
45 decreases(length - i))
46 {
47 data[i] = state_ptr[i];
48 }
49#else /* MLD_SYS_LITTLE_ENDIAN */
50 /* Portable version */
51 for (i = 0; i < length; i++)
52 __loop__(invariant(i <= length)
53 decreases(length - i))
54 {
55 data[i] = (state[(offset + i) >> 3] >> (8 * ((offset + i) & 0x07))) & 0xFF;
56 }
57#endif /* !MLD_SYS_LITTLE_ENDIAN */
58}

60MLD_INTERNAL_API
61void mld_keccakf1600_xor_bytes(uint64_t *state, const unsigned char *data,
62 unsigned offset, unsigned length)
63{
64 unsigned i;
65#if defined(MLD_SYS_LITTLE_ENDIAN)
66 uint8_t *state_ptr = (uint8_t *)state + offset;
67 for (i = 0; i < length; i++)
68 __loop__(invariant(i <= length)
69 decreases(length - i))
70 {
71 state_ptr[i] ^= data[i];
72 }
73#else /* MLD_SYS_LITTLE_ENDIAN */
74 /* Portable version */
75 for (i = 0; i < length; i++)
76 __loop__(invariant(i <= length)
77 decreases(length - i))
78 {
79 state[(offset + i) >> 3] ^= (uint64_t)data[i]
80 << (8 * ((offset + i) & 0x07));
81 }
82#endif /* !MLD_SYS_LITTLE_ENDIAN */
83}
