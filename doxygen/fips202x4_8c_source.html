27static void mld_keccak_absorb_once_x4(uint64_t *s, unsigned r,
28 const uint8_t *in0, const uint8_t *in1,
29 const uint8_t *in2, const uint8_t *in3,
30 size_t inlen, uint8_t p)
31__contract__(
32 requires(inlen <= MLD_MAX_BUFFER_SIZE)
33 requires(memory_no_alias(s, sizeof(uint64_t) * MLD_KECCAK_LANES * MLD_KECCAK_WAY))
34 requires(r > 0)
35 requires(r <= sizeof(uint64_t) * MLD_KECCAK_LANES)
36 requires(memory_no_alias(in0, inlen))
37 requires(memory_no_alias(in1, inlen))
38 requires(memory_no_alias(in2, inlen))
39 requires(memory_no_alias(in3, inlen))
40 assigns(memory_slice(s, sizeof(uint64_t) * MLD_KECCAK_LANES * MLD_KECCAK_WAY)))
41{
42 while (inlen >= r)
43 __loop__(
44 assigns(inlen, in0, in1, in2, in3, memory_slice(s, sizeof(uint64_t) * MLD_KECCAK_LANES * MLD_KECCAK_WAY))
45 invariant(inlen <= loop_entry(inlen))
46 invariant(in0 == loop_entry(in0) + (loop_entry(inlen) - inlen))
47 invariant(in1 == loop_entry(in1) + (loop_entry(inlen) - inlen))
48 invariant(in2 == loop_entry(in2) + (loop_entry(inlen) - inlen))
49 invariant(in3 == loop_entry(in3) + (loop_entry(inlen) - inlen))
50 decreases(inlen))
51 {
52 mld_keccakf1600x4_xor_bytes(s, in0, in1, in2, in3, 0, r);
53 mld_keccakf1600x4_permute(s);
54
55 in0 += r;
56 in1 += r;
57 in2 += r;
58 in3 += r;
59 inlen -= r;
60 }
61
62 /* Safety: At this point, inlen < r, so the truncations to unsigned are safe
63 * below. */
64 if (inlen > 0)
65 {
66 mld_keccakf1600x4_xor_bytes(s, in0, in1, in2, in3, 0, (unsigned)inlen);
67 }
68
69 if (inlen == r - 1)
70 {
71 p |= 128;
72 mld_keccakf1600x4_xor_bytes(s, &p, &p, &p, &p, (unsigned)inlen, 1);
73 }
74 else
75 {
76 mld_keccakf1600x4_xor_bytes(s, &p, &p, &p, &p, (unsigned)inlen, 1);
77 p = 128;
78 mld_keccakf1600x4_xor_bytes(s, &p, &p, &p, &p, r - 1, 1);
79 }
80}

82static void mld_keccak_squeezeblocks_x4(uint8_t *out0, uint8_t *out1,
83 uint8_t *out2, uint8_t *out3,
84 size_t nblocks, uint64_t *s, unsigned r)
85__contract__(
86 requires(r <= sizeof(uint64_t) * MLD_KECCAK_LANES)
87 requires(nblocks <= 8 /* somewhat arbitrary bound */)
88 requires(memory_no_alias(s, sizeof(uint64_t) * MLD_KECCAK_LANES * MLD_KECCAK_WAY))
89 requires(memory_no_alias(out0, nblocks * r))
90 requires(memory_no_alias(out1, nblocks * r))
91 requires(memory_no_alias(out2, nblocks * r))
92 requires(memory_no_alias(out3, nblocks * r))
93 assigns(memory_slice(s, sizeof(uint64_t) * MLD_KECCAK_LANES * MLD_KECCAK_WAY))
94 assigns(memory_slice(out0, nblocks * r))
95 assigns(memory_slice(out1, nblocks * r))
96 assigns(memory_slice(out2, nblocks * r))
97 assigns(memory_slice(out3, nblocks * r)))
98{
99 size_t current_offset = 0;
100 while (nblocks > 0)
101 __loop__(
102 assigns(nblocks, current_offset,
103 memory_slice(s, sizeof(uint64_t) * MLD_KECCAK_LANES * MLD_KECCAK_WAY),
104 memory_slice(out0, nblocks * r),
105 memory_slice(out1, nblocks * r),
106 memory_slice(out2, nblocks * r),
107 memory_slice(out3, nblocks * r))
108 invariant(nblocks <= loop_entry(nblocks))
109 invariant(current_offset == (loop_entry(nblocks) - nblocks) * r)
110 decreases(nblocks))
111 {
112 mld_keccakf1600x4_permute(s);
113 mld_keccakf1600x4_extract_bytes(
114 s, &out0[current_offset], &out1[current_offset], &out2[current_offset],
115 &out3[current_offset], 0, r);
116 current_offset += r;
117 nblocks--;
118 }
119}
