70/* Size of TR in bytes (level-independent)
71 *
72 * TR = SHAKE256(pk, 64) is the hash of the public key. Callers of the
73 * external-mu API (signature_extmu / verify_extmu) that compute the message
74 * representative mu = SHAKE256(TR || M', MLDSA_CRHBYTES) themselves -- e.g. to
75 * sign or verify a message that is too large or streamed to hold in memory --
76 * need this constant to size the TR buffer. */
77#define MLDSA_TRBYTES 64
78#define MLDSA44_TRBYTES MLDSA_TRBYTES
79#define MLDSA65_TRBYTES MLDSA_TRBYTES
80#define MLDSA87_TRBYTES MLDSA_TRBYTES

85#define MLDSA65_RNDBYTES MLDSA_RNDBYTES
86#define MLDSA87_RNDBYTES MLDSA_RNDBYTES
